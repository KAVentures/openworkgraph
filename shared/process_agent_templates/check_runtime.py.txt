"""Run synthetic acceptance on a trusted, externally pinned OWG runtime ZIP.

This is not an OS sandbox. Run only trusted exports, in an isolated Python
environment, without customer credentials. No real-case or provider acceptance
is inferred from these bundled tests.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile

from verify_release import ReleaseIntegrityError, verify_directory

FORMAT = "openworkgraph.customer-runtime-synthetic-acceptance.v1"
REQUIRED = ("test_shadow", "test_agent_sdk", "test_execution", "test_monitor",
            "test_configuration", "test_evaluation", "test_binding_rehearsal", "test_source_planning",
            "test_source_evaluation", "test_run_queue", "test_execute_http")
OPTIONAL = ("test_decisions",)
TIMEOUT = 120
_CHILD = """
import json, pathlib, sys, unittest, os
# Isolated interpreter mode excludes the working directory from sys.path.
sys.path.insert(0, os.getcwd())
name = sys.argv[1]
suite = unittest.defaultTestLoader.loadTestsFromName(name)
class QuietResult(unittest.TestResult):
    pass
result = QuietResult()
suite.run(result)
summary = dict(tests_run=result.testsRun, failures=len(result.failures),
               errors=len(result.errors), skipped=len(result.skipped),
               expected_failures=len(result.expectedFailures),
               unexpected_successes=len(result.unexpectedSuccesses))
pathlib.Path(".owg-acceptance-counts.json").write_text(json.dumps(summary), encoding="utf-8")
"""


def _environment(work: Path) -> dict[str, str]:
    # Use the current interpreter and installed dependencies, but do not copy
    # provider credentials, proxy configuration, PYTHONPATH or customer HOME.
    env = {name: os.environ[name] for name in
           ("PATH", "SYSTEMROOT", "SystemRoot", "WINDIR", "COMSPEC", "PATHEXT")
           if name in os.environ}
    env.update(HOME=str(work), USERPROFILE=str(work), APPDATA=str(work),
               LOCALAPPDATA=str(work), TMPDIR=str(work), TEMP=str(work), TMP=str(work),
               OPENAI_AGENTS_DISABLE_TRACING="1")
    return env


def _run_suite(module: str, root: Path, home: Path) -> dict:
    result_path = root / ".owg-acceptance-counts.json"
    result_path.unlink(missing_ok=True)
    try:
        completed = subprocess.run(
            [sys.executable, "-I", "-B", "-c", _CHILD, module],
            cwd=root, env=_environment(home), stdin=subprocess.DEVNULL,
            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
            timeout=TIMEOUT, check=False)
    except subprocess.TimeoutExpired:
        return {"suite": module, "status": "timed_out", "counts": None}
    except OSError:
        return {"suite": module, "status": "runner_error", "counts": None}
    if completed.returncode != 0 or not result_path.is_file():
        return {"suite": module, "status": "runner_error", "counts": None}
    try:
        with result_path.open("rb") as handle:
            raw = handle.read(4097)
        if len(raw) > 4096:
            raise ValueError
        counts = json.loads(raw)
        keys = {"tests_run", "failures", "errors", "skipped",
                "expected_failures", "unexpected_successes"}
        if not isinstance(counts, dict) or set(counts) != keys or any(
                type(v) is not int or v < 0 for v in counts.values()):
            raise ValueError
    except (OSError, ValueError):
        return {"suite": module, "status": "runner_error", "counts": None}
    if counts["failures"] or counts["errors"] or counts["unexpected_successes"]:
        status = "failed"
    elif not counts["tests_run"] or counts["skipped"] or counts["expected_failures"]:
        status = "incomplete"
    else:
        status = "passed"
    return {"suite": module, "status": status, "counts": counts}


def acceptance(root: Path, pinned_digest: str) -> dict:
    """Verify the independent pin, then run only generated synthetic test modules.

    All manifested bytes are recopied and reverified in a private temporary
    directory. Customer-added files are not imported or passed to test code.
    """
    if not isinstance(pinned_digest, str) or len(pinned_digest) != 64:
        raise ReleaseIntegrityError("An independently recorded manifest SHA-256 is required")
    verified = verify_directory(root, expected_manifest_sha256=pinned_digest)
    raw = (root / "release-manifest.json").read_bytes()
    if hashlib.sha256(raw).hexdigest() != pinned_digest:
        raise ReleaseIntegrityError("Manifest changed after verification")
    manifest = json.loads(raw)
    hashes = manifest["files"]
    with tempfile.TemporaryDirectory(prefix="owg-runtime-acceptance-") as directory:
        work = Path(directory)
        staged = work / "release"
        staged.mkdir()
        home = work / "home"
        home.mkdir()
        (staged / "release-manifest.json").write_bytes(raw)
        for name in hashes:
            source = root / name
            with source.open("rb") as handle:
                data = handle.read(4 * 1024 * 1024 + 1)
            if hashlib.sha256(data).hexdigest() != hashes[name]:
                raise ReleaseIntegrityError("Generated file changed after verification")
            target = staged / name
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(data)
        verify_directory(staged, expected_manifest_sha256=pinned_digest)
        spec = json.loads((staged / "workflow.spec.json").read_bytes())
        modules = list(REQUIRED)
        if spec.get("process_model", {}).get("decision_tables"):
            modules += OPTIONAL
        if any(module + ".py" not in hashes for module in modules):
            raise ReleaseIntegrityError("Required synthetic acceptance suite missing from the manifest")
        suites = [_run_suite(module, staged, home) for module in modules]
    statuses = {s["status"] for s in suites}
    status = ("failed" if statuses & {"failed", "runner_error", "timed_out"}
              else "incomplete" if statuses != {"passed"} else "passed")
    return {
        "format": FORMAT, "workflow_reference": verified["workflow_reference"],
        "manifest_sha256": pinned_digest,
        "status": status, "suites": suites,
        "generated_file_integrity_ok": True,
        "synthetic_only": True, "customer_files_included": False,
        "customer_credentials_inherited": False,
        "os_sandbox_enforced": False, "independent_signature_verified": False,
        "real_case_quality_verified": False, "remote_deployment_verified": False,
        "business_outcomes_verified": False, "execution_authorized": False,
    }


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root", type=Path, default=Path("."))
    parser.add_argument("--manifest-sha256", required=True)
    parser.add_argument("--confirm-trusted-synthetic-tests", action="store_true")
    args = parser.parse_args(argv)
    if not args.confirm_trusted_synthetic_tests:
        parser.error("Explicitly confirm running synthetic code from a trusted export")
    try:
        result = acceptance(args.root, args.manifest_sha256)
    except (ReleaseIntegrityError, OSError, ValueError):
        print("Acceptance refused: invalid, missing or changed release files.", file=sys.stderr)
        return 2
    print(json.dumps(result, sort_keys=True, indent=2))
    return {"passed": 0, "incomplete": 3, "failed": 1}[result["status"]]


if __name__ == "__main__":
    raise SystemExit(main())
