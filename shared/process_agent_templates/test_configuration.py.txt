    async def test_route_inventory_randomized_intersection(self):
        from configure import _reviewed_route_inventory
        from random import Random
        rng=Random(20261124)
        pool=[['start',str(i),'end'] for i in range(8)]
        for _ in range(1000):
            cases=[{'id':f'case_{i}','input':{'v':rng.randrange(5)},
                    'expected':{'accepted_routes':rng.sample(pool,rng.randrange(5))}}
                   for i in range(rng.randrange(1,21))]
            target={'v':rng.randrange(6)}
            inventory,exact,common=_reviewed_route_inventory(cases,digest(target))
            matches=[row for row in cases if row['input']==target]
            self.assertEqual(exact,matches)
            self.assertEqual(common,{tuple(route) for route in pool if matches and all(
                route in row['expected']['accepted_routes'] for row in matches)})
            self.assertEqual(set(inventory),{tuple(route) for row in cases for route in row['expected']['accepted_routes']})
            for route,item in inventory.items():
                self.assertEqual(item['references'],[
                    {'case_id':row['id'],'alternative':i+1} for row in cases
                    for i, candidate in enumerate(row['expected']['accepted_routes']) if tuple(candidate)==route])
                self.assertEqual(item['exact_references'],[
                    {'case_id':row['id'],'alternative':i+1} for row in matches
                    for i, candidate in enumerate(row['expected']['accepted_routes']) if tuple(candidate)==route])

    async def test_paired_audit_delta_rejects_mismatched_bases_and_matches_randomized_oracle(self):

        rng=Random(20261124)

               'catalog_sha256':'c','suite_sha256':'t'}
        universe=[(code,activity) for code in ('pin','missing','type','auth') for activity in ('a','b',None)]
        for _ in range(100):
            old,new=(set(rng.sample(universe,rng.randrange(len(universe)+1))) for _ in range(2))

            self.assertEqual({(f['code'],f['activity_id']) for f in delta['findings_added']},new-old)
            self.assertEqual({(f['code'],f['activity_id']) for f in delta['findings_removed']},old-new)
            self.assertEqual(delta['retained_finding_count'],len(old&new))
            self.assertEqual(delta['disposition'],{

                (True,True):'unchanged_clear',(False,False):'unchanged_findings'}[not old,not new])

            with self.assertRaises(ImplementationError):
                _paired_audit_delta(a,{**b,field:'different'})

            _paired_audit_delta(a,{**b,'cases':[]})
