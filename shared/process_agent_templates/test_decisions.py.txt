def receipt(request,state='open',when=None):
    from process_source_state import state_receipt
    return state_receipt(request['resource_ref'],request['source_type'],
        {'source':{'provider':'github','record_kind':'pull_request','state':state}},
        (when or datetime.now(timezone.utc)).isoformat(),'source_state_reported',True)


def intent_for(data):
    spec,plan,case,bindings=data
    result=review_template(spec,prepare(*data),case,bindings)
    result.update(customer_authorization_confirmed=True,source_reads_confirmed=True,
                  expires_at=(datetime.now(timezone.utc)+timedelta(minutes=10)).isoformat())
    return result

class DecisionExecution(unittest.IsolatedAsyncioTestCase):

        self.data=fixture();self.tools=FakeTools();self.reads=[];self.state='open';self.mutation={}
        self.folder=TemporaryDirectory();self.addCleanup(self.folder.cleanup)
        env=patch.dict('os.environ',{'OWG_EXECUTION_MCP_TOKEN':'SYNTHETIC_SECRET',
                                    'OWG_EXECUTION_SOURCE_READ_TOKEN':'SYNTHETIC_SOURCE_SECRET'})
        env.start();self.addCleanup(env.stop)
    async def source(self,base,token,request):
        self.reads.append(deepcopy(request));return receipt(request,self.state)|self.mutation
    async def outcomes(self,base,token,request):
        value=outcome_receipt(request)
        if not self.tools.resolved:value.update(operation_verified=False,availability='unknown',reason='provider_state_not_expected_operation')
        return value
    async def run_case(self,**options):
        return await execute_approved(*self.data,intent_for(self.data),journal_directory=self.folder.name,
            client_factory=lambda *args:self.tools,source_state_reader=self.source,receipt_reader=self.outcomes,**options)
    async def test_matching_route_rechecks_before_precondition_every_action_and_completion(self):
        result=await self.run_case()
        self.assertEqual([c[0] for c in self.tools.calls],['read_case','resolve_case','read_case'])
        self.assertEqual(len(self.reads),5)
        self.assertTrue(result['declared_route_checks_satisfied'])
        from monitor import report,read_journal
        monitoring=report(self.data[0],read_journal(next(Path(self.folder.name).glob('*.jsonl'))),'Synthetic decision runtime')
        self.assertEqual(monitoring['runs'][0]['status'],'succeeded')
        self.assertIsNone(monitoring['runs'][0]['outcome']['verified'])
        self.assertTrue(result['all_declared_operations_verified'])
        self.assertFalse(result['business_objective_verified']);self.assertFalse(result['agent_causality_verified'])
        text=''.join(p.read_text() for p in Path(self.folder.name).glob('*.jsonl'))
        for private in ('SYNTHETIC_SECRET','SYNTHETIC_SOURCE_SECRET',self.data[2]['record_ref'],'Synthetic unverified policy'):
            self.assertNotIn(private,text)
    async def test_alternative_gap_unavailable_and_stale_stop_before_tools(self):
        for state,changes in [('merged',{}),('closed_unmerged',{}),('open',{'source_state_verified':False}),
                ('open',{'verified_at':'2000-01-01T00:00:00Z'})]:
            self.state=state;self.mutation=changes
            with TemporaryDirectory() as folder:
                result=await execute_approved(*self.data,intent_for(self.data),journal_directory=folder,
                    client_factory=lambda *args:self.tools,source_state_reader=self.source)
                self.assertTrue(result['stopped_before_tools'])
        self.assertFalse(self.tools.calls)
    async def test_changed_state_after_write_precondition_prevents_write(self):
        original=self.tools.call_tool
        async def change(*args):
            value=await original(*args);self.state='merged';return value
        self.tools.call_tool=change
        result=await self.run_case()
        self.assertEqual([c[0] for c in self.tools.calls],['read_case'])
        self.assertFalse(result['declared_route_checks_satisfied'])
        self.assertFalse(result['all_tool_results_matched'])
    async def test_effective_date_expiring_during_tool_inventory_prevents_even_precondition(self):
        import decision_runtime
        day=datetime.now(timezone.utc).date()
        table=self.data[0]['process_model']['decision_tables'][0];table['effective_until']=(day+timedelta(days=1)).isoformat()
        self.data[0]['process_model_sha256']=digest(self.data[0]['process_model'])
        clock=[datetime.now(timezone.utc)]
        class Clock:
            @staticmethod
            def now(*args):return clock[0]
        original=self.tools.list_tools
        async def expire():
            clock[0]+=timedelta(days=1);return await original()
        self.tools.list_tools=expire

        self.assertFalse(self.tools.calls)
        self.assertFalse(result['declared_route_checks_satisfied'])
        self.assertEqual(result['decision_table_checks'][-1]['status'],'expired')

class DecisionPlanning(unittest.IsolatedAsyncioTestCase):
    asyncSetUp=DecisionExecution.asyncSetUp
    source=DecisionExecution.source

    async def propose(self,*,choose=False,advance=False,**options):
        from test_agent_sdk import OfflineModel
        from plan_sources import propose_with_sources
        spec,plan,case,bindings=self.data
        case['text']='SYNTHETIC_UNTRUSTED_INSTRUCTION';plan['missing_business_inputs']=[]
        owner=self;clock=[datetime.now(timezone.utc)]
        class Model(OfflineModel):
            async def get_response(self,*args,**kwargs):
                raw=kwargs.get('input') if 'input' in kwargs else args[1]
                payload=json.loads(raw if isinstance(raw,str) else raw[0]['content'])
                owner.payload=payload
                table=payload['current_source_decision_checks'][0]
                assert set(table)=={'table_id','status','selected_target_activity_id'}
                assert 'provider_reported_state' not in json.dumps(payload)
                if choose and table['selected_target_activity_id']=='hold':self.plan={**self.plan,'activity_ids':['choose','hold']}
                if advance:clock[0]+=timedelta(seconds=121)
                return await super().get_response(*args,**kwargs)
        return await propose_with_sources(case,bindings,model=Model(plan),spec=spec,source_state_reader=self.source,
            source_reads_confirmed=True,share_rule_checks_with_model_confirmed=True,clock=lambda:clock[0],**options)
    async def test_actual_sdk_selects_alternative_from_minimized_snapshot_and_rejects_model_disagreement(self):
        for state in ('open','merged'):
            self.state=state
            result=await self.propose(choose=True)
            self.assertEqual(result['plan']['status'],'proposal')
            self.assertEqual(result['source_planning']['source_receipt_request_count'],1)
            self.assertFalse(result['execution_permitted']);self.assertFalse(self.tools.calls)
        result=await self.propose()
        self.assertEqual(result['plan']['status'],'needs_clarification')
        result=await self.propose(choose=True,advance=True)
        self.assertEqual(result['plan']['status'],'needs_clarification')
    async def test_offline_planner_caller_facts_cannot_substitute_for_source_table_context(self):
        from agent import propose
        from test_agent_sdk import OfflineModel
        spec,plan,case,bindings=self.data
        case.update(source={'state':'open'},text='SYNTHETIC_UNTRUSTED_INSTRUCTION');plan['missing_business_inputs']=[]
        result=await propose(case,model=OfflineModel(plan),spec=spec)
