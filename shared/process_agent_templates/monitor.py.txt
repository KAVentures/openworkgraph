"""Offline projection of one exact-spec customer journal into an OWG run report.

Imported journals remain customer reports, including their source-check claims.
This module never connects to OWG, a model provider or a business tool.
"""
import argparse
from collections import Counter
import hashlib
from datetime import datetime, timezone
import json
import os
from pathlib import Path
import re

from execute import ImplementationError, _pairs, _constant, digest, load_object, reference
from runtime import validate_spec, check_plan

MAX_BYTES = 8 * 1024 * 1024
MAX_RECORDS = 10000
MAX_BATCH_JOURNALS = 100
MAX_BATCH_BYTES = 64 * 1024 * 1024
MAX_EXPORT_BYTES = 2 * 1024 * 1024
_PHASES = {
    'started', 'catalog_checked', 'source_rule_checked', 'decision_table_checked', 'stopped_before_tools',
    'source_preflight', 'precondition_pending', 'precondition_not_established',
    'attempt_pending', 'tool_returned_expected_structure', 'tool_reported_failure',
    'uncertain', 'implementation_contract_or_intent_not_established',
    'transport_or_tool_inventory_unavailable', 'source_rules_not_established_at_completion',
    'source_checked', 'finished', 'operator_stop_observed', 'operator_stop_active_at_completion',
}
_STOPS = {'implementation_contract_or_intent_not_established',
          'transport_or_tool_inventory_unavailable', 'source_rules_not_established_at_completion',
          'precondition_not_established'}


def _fail():
    raise ImplementationError('Select a complete, bounded journal for this exact exported specification')


def _stamp(value):
    if not isinstance(value, str) or len(value) > 64:
        _fail()
    try:
        stamp = datetime.fromisoformat(value.replace('Z', '+00:00'))
        if stamp.tzinfo is None or stamp.utcoffset() is None:
            _fail()
        return stamp.astimezone(timezone.utc)
    except (ValueError, OverflowError):
        _fail()


def _read_snapshot(path, limit=MAX_BYTES):
    with Path(path).open('rb') as stream:
        content = stream.read(limit + 1)
    return content


def _decode_journal(content):
    if not content or len(content) > MAX_BYTES or not content.endswith(b'\n'):
        _fail()  # A torn trailing record must not become a completed run.
    lines = content.splitlines()
    if not 1 <= len(lines) <= MAX_RECORDS:
        _fail()
    try:
        return [json.loads(line, object_pairs_hook=_pairs, parse_constant=_constant) for line in lines]
    except (ValueError, UnicodeError, RecursionError):
        _fail()


def read_journal(path):
    return _decode_journal(_read_snapshot(path))


def report(spec, records, deployment_name):
    validate_spec(spec)
    if (not isinstance(deployment_name, str) or not 1 <= len(deployment_name.strip()) <= 120
            or any(ord(c) < 32 or 127 <= ord(c) <= 159 for c in deployment_name)):
        raise ImplementationError('Use a short deployment label without control characters')
    if not isinstance(records, list) or not 1 <= len(records) <= MAX_RECORDS:
        _fail()
    first = records[0]
    if (not isinstance(first, dict) or first.get('phase') != 'started'
            or first.get('spec_sha256') != digest(spec)
            or first.get('process_reference') != reference(spec)
            or not isinstance(first.get('execution_id'), str)
            or not re.fullmatch(r'[0-9a-f]{64}', first['execution_id'])):
        _fail()
    nodes = {node['id'] for node in spec['process_model']['activities']
             if node['kind'] == 'activity' and node.get('implementation') != 'human'}
    dependent=spec.get('decision_table_contract')=='openworkgraph.current-source-dependent-decision.v1'
    expected_routes={};expected_order=[];checkpoint=0;checkpoint_rows={};checkpoint_ready=False
    declared_actions=[];declared_preconditions=set();preconditions_seen=set()
    checkpoint_kind=None;checkpoint_attempt=None;initial_checked=False;completion_checked=False
    expected_source_rules=[];source_checks={};source_number=0;source_key=None;source_consumed=set()
    checkpoint_times={};last_checkpoint_time=None
    if dependent:
        from owg_model.process_model import ProcessModel
        from owg_model.process_decision_tables import decision_order
        from decision_runtime import _required_tables
        model=ProcessModel.model_validate(spec['process_model'])
        tables_by_id={t.id:t for t in model.decision_tables}
        declared=first.get('dependent_decision_routes')
        if not isinstance(declared,list) or len(declared)>12:
            _fail()
        for route in declared:
            if (not isinstance(route,dict) or set(route)!={'table_id','target_activity_ids'}
                or not isinstance(route['table_id'],str) or route['table_id'] not in tables_by_id
                or route['table_id'] in expected_routes or not isinstance(route['target_activity_ids'],list)
                or len(route['target_activity_ids'])>80):
                _fail()
            table=tables_by_id[route['table_id']]
            if any(not isinstance(t,str) or t not in {r.target_activity_id for r in table.rows}
                   for t in route['target_activity_ids']):
                _fail()
            expected_routes[table.id]=route['target_activity_ids']
        path=first.get('declared_activity_ids')
        plan_hash=first.get('declared_plan_sha256')
        if not isinstance(plan_hash,str) or re.fullmatch(r'[0-9a-f]{64}',plan_hash) is None:
            _fail()
        try:
            structural=check_plan({'revision':spec['revision'],'content_sha256':spec['content_sha256'],
                'activity_ids':path},spec)
        except (ValueError,TypeError,KeyError):
            _fail()
        route_targets={r['table_id']:r['target_activity_ids'] for r in structural.get('decision_table_routes',[])}
        if {k:v for k,v in expected_routes.items() if v}!=route_targets:
            _fail()
        declared_actions=[ident for ident in path if ident in nodes]
        expected_source_rules=sorted(structural.get('non_table_structured_rule_ids',[]))
        if first.get('declared_source_rule_ids')!=expected_source_rules or structural['unconfirmed_rule_ids']:
            _fail()
        preconditions=first.get('declared_precondition_attempts')
        if (not isinstance(preconditions,list) or any(type(i) is not int or not 1<=i<=len(declared_actions) for i in preconditions)
            or preconditions!=sorted(set(preconditions))):
            _fail()
        declared_preconditions=set(preconditions)
        visited={k for k,v in expected_routes.items() if v}
        if _required_tables(model,visited)!=set(expected_routes):
            _fail()
        expected_order=[t.id for t in decision_order(model.decision_tables) if t.id in expected_routes]
        if [r['table_id'] for r in declared]!=expected_order:
            _fail()
    started = previous = _stamp(first.get('at'))
    steps, pending, ended, blocked = [], None, None, False
    attempts = 0
    dependency_pin=spec.get('dependency_build_basis_sha256')
    dependency_required='dependency_build_basis_sha256' in spec
    if dependency_required and (not isinstance(dependency_pin,str) or not re.fullmatch(r'[a-f0-9]{64}',dependency_pin)
                                or not spec.get('organization_catalog_reference')):
        _fail()
    dependency_checks=0;dependency_initial=False;dependency_completion=False
    precondition_attempts=set();expected_actions=[];expected_preconditions=set()
    if dependency_required:
        route=first.get('dependency_route_activity_ids')
        declared=first.get('dependency_precondition_attempts')
        try:
            check_plan({'revision':spec['revision'],'content_sha256':spec['content_sha256'],
                'activity_ids':route},spec)
        except (ValueError,TypeError,KeyError):
            _fail()
        expected_actions=[node for node in route if node in nodes]
        if (not isinstance(declared,list) or any(type(i) is not int or not 1<=i<=len(expected_actions) for i in declared)
            or declared!=sorted(set(declared))):
            _fail()
        expected_preconditions=set(declared)
    for index, row in enumerate(records[1:], 1):
        if (not isinstance(row, dict) or not isinstance(row.get('phase'), str)
                or row['phase'] not in _PHASES or row['phase'] == 'started' or ended):
            _fail()
        stamp = _stamp(row.get('at'))
        if stamp < previous:
            _fail()
        previous = stamp
        phase = row['phase']
        if dependent and phase in {'source_rule_checked','decision_table_checked'}:
            purpose=row.get('checkpoint_purpose');attempt=row.get('checkpoint_attempt')
            if not isinstance(purpose,str) or (attempt is not None and type(attempt) is not int):
                _fail()
            key=(purpose,attempt)
            checked_at=_stamp(row.get('checkpoint_checked_at'))
            if key in checkpoint_times:
                if checkpoint_times[key]!=checked_at:
                    _fail()
            else:
                if last_checkpoint_time is not None and checked_at<last_checkpoint_time:
                    _fail()
                checkpoint_times[key]=checked_at;last_checkpoint_time=checked_at
            if phase=='decision_table_checked' and row.get('checked_on_utc')!=checked_at.date().isoformat():
                _fail()
        if dependent and expected_source_rules and phase=='source_rule_checked':
            number=row.get('checkpoint_number');purpose=row.get('checkpoint_purpose');attempt=row.get('checkpoint_attempt')
            if type(number) is not int or not isinstance(purpose,str) or purpose not in {'initial','precondition','action','completion'}:
                _fail()
            if purpose in {'initial','completion'}:
                if attempt is not None or (purpose=='initial' and (attempts or number!=1)):
                    _fail()
            elif type(attempt) is not int or attempt!=attempts+1 or attempt>len(declared_actions) or pending:
                _fail()
            key=(purpose,attempt)
            if number==source_number+1:
                if key in source_checks or (source_key is not None and len(source_checks[source_key])!=len(expected_source_rules)):
                    _fail()
                source_number=number;source_key=key;source_checks[key]=[]
            if number!=source_number or key!=source_key or key in source_consumed:
                _fail()
            rows=source_checks[key]
            if len(rows)>=len(expected_source_rules) or row.get('rule_id')!=expected_source_rules[len(rows)]:
                _fail()
            if row.get('status') not in {'matched','not_matched','unknown'} or row.get('record_association')!='customer_selected_not_independently_established':
                _fail()
            rows.append(row['status'])
        elif dependent and phase=='source_rule_checked':
            _fail()
        if dependent and expected_order and phase!='decision_table_checked':
            if checkpoint_rows and len(checkpoint_rows)!=len(expected_order):
                _fail()
            if phase in {'precondition_pending','attempt_pending'}:
                purpose='precondition' if phase=='precondition_pending' else 'action'
                if (not initial_checked or not checkpoint_ready or blocked or completion_checked
                    or checkpoint_kind!=purpose or checkpoint_attempt!=attempts+1):
                    _fail()
                checkpoint_ready=False
        if dependent and phase in {'precondition_pending','attempt_pending'}:
            if expected_source_rules:
                initial=('initial',None)
                key=('precondition' if phase=='precondition_pending' else 'action',attempts+1)
                if (source_checks.get(initial)!=['matched']*len(expected_source_rules)
                    or source_checks.get(key)!=['matched']*len(expected_source_rules) or key in source_consumed
                    or source_key!=key):
                    _fail()
                source_consumed.add(key)
            if (attempts>=len(declared_actions) or row.get('activity_id')!=declared_actions[attempts]
                or type(row.get('attempt')) is not int or row['attempt']!=attempts+1):
                _fail()
            if phase=='precondition_pending':
                if attempts+1 not in declared_preconditions or attempts+1 in preconditions_seen or pending:
                    _fail()
                preconditions_seen.add(attempts+1)
            elif (attempts+1 in declared_preconditions)!=(attempts+1 in preconditions_seen):
                _fail()
        if dependency_required and phase=='catalog_checked':
            purpose=row.get('dependency_check_purpose');attempt=row.get('dependency_check_attempt')
            number=row.get('dependency_check_number')
            current=row.get('current_independent_design_review')
            if (row.get('dependency_build_basis_sha256')!=dependency_pin or type(number) is not int
                or number!=dependency_checks+1 or not isinstance(purpose,str)
                or purpose not in {'initial','refresh','precondition','action','completion'}
                or type(current) is not bool or dependency_completion or pending):
                _fail()
            if purpose in {'precondition','action'}:
                if type(attempt) is not int or attempt!=attempts+1 or blocked or not dependency_initial:
                    _fail()
            elif attempt is not None:
                _fail()
            if purpose=='initial':
                if dependency_checks or index!=1:
                    _fail()
                dependency_initial=current
            elif not dependency_checks:
                _fail()
            if purpose=='completion':
                dependency_completion=True
            dependency_checks=number
        if dependency_required and phase in {'precondition_pending','attempt_pending'}:
            purpose='precondition' if phase=='precondition_pending' else 'action'
            prior=records[index-1]
            if (not dependency_initial or dependency_completion or blocked or pending
                or prior.get('phase')!='catalog_checked'
                or prior.get('current_independent_design_review') is not True
                or prior.get('dependency_check_purpose')!=purpose
                or type(row.get('attempt')) is not int or row['attempt']!=attempts+1
                or prior.get('dependency_check_attempt')!=row['attempt']):
                _fail()
            if purpose=='precondition':
                if (row['attempt'] in precondition_attempts or row['attempt'] not in expected_preconditions
                    or not isinstance(row.get('activity_id'),str) or row['activity_id'] not in nodes
                    or row['attempt']>len(expected_actions) or row['activity_id']!=expected_actions[row['attempt']-1]):
                    _fail()
                precondition_attempts.add(row['attempt'])
            elif (row['attempt']>len(expected_actions) or row.get('activity_id')!=expected_actions[row['attempt']-1]
                  or (row['attempt'] in expected_preconditions and row['attempt'] not in precondition_attempts)):
                _fail()
        if phase == 'operator_stop_observed':
            if (row.get('future_tool_attempts_stopped') is not True
                    or row.get('already_in_flight_requests_not_cancelled') is not True):
                _fail()
            blocked = True
        elif phase == 'operator_stop_active_at_completion':
            if row.get('in_flight_requests_not_cancelled') is not True:
                _fail()
            blocked = True
        if phase == 'attempt_pending':
            if (blocked or pending is not None or not isinstance(row.get('activity_id'), str) or row['activity_id'] not in nodes
                    or type(row.get('attempt')) is not int or row['attempt'] != attempts + 1
                    or attempts >= 80):
                _fail()
            attempts += 1
            pending = (row['activity_id'], row['attempt'], stamp)
        elif phase in {'tool_returned_expected_structure', 'tool_reported_failure', 'uncertain'}:
            if (pending is None or type(row.get('attempt')) is not int
                    or (row.get('activity_id'), row.get('attempt')) != pending[:2]):
                _fail()
            status = {'tool_returned_expected_structure': 'completed', 'tool_reported_failure': 'failed',
                      'uncertain': 'handed_to_human'}[phase]
            steps.append({'step_id': pending[0], 'activity': pending[0], 'status': status, 'at': stamp.isoformat()})
            blocked = blocked or status != 'completed'
            pending = None
        elif phase == 'precondition_not_established':
            if (not isinstance(row.get('activity_id'), str) or row['activity_id'] not in nodes
                    or type(row.get('attempt')) is not int or row['attempt'] != attempts + 1 or pending):
                _fail()
        if phase in _STOPS or (phase == 'source_rule_checked' and row.get('status') != 'matched'):
            blocked = True
        if phase == 'decision_table_checked':
            tables={t['id']:t for t in spec['process_model'].get('decision_tables',[])}
            if (not isinstance(row.get('table_id'),str) or row['table_id'] not in tables
                    or type(row.get('route_matches')) is not bool
                    or row.get('execution_authorized') is not False
                    or row.get('policy_document_verified') is not False):
                _fail()
            table=tables[row['table_id']];first_policy=table.get('hit_policy','UNIQUE')=='FIRST'
            if dependent:
                ident=table['id']
                if not expected_order or ident not in expected_routes:
                    _fail()
                number=row.get('checkpoint_number')
                if type(number) is not int:
                    _fail()
                if number==checkpoint+1:
                    if checkpoint_rows and len(checkpoint_rows)!=len(expected_order):
                        _fail()
                    checkpoint=number;checkpoint_rows={};checkpoint_ready=False
                    checkpoint_kind=row.get('checkpoint_purpose')
                    checkpoint_attempt=row.get('checkpoint_attempt')
                    if expected_source_rules:
                        if not isinstance(checkpoint_kind,str) or (checkpoint_attempt is not None and type(checkpoint_attempt) is not int):
                            _fail()
                        key=(checkpoint_kind,checkpoint_attempt)
                        if key!=source_key or source_checks.get(key)!=['matched']*len(expected_source_rules):
                            _fail()
                    if checkpoint_kind=='initial':
                        if checkpoint!=1 or checkpoint_attempt is not None:
                            _fail()
                    elif checkpoint_kind=='completion':
                        if not initial_checked or completion_checked or checkpoint_attempt is not None or pending:
                            _fail()
                    elif checkpoint_kind in {'precondition','action'}:
                        if (not initial_checked or completion_checked or pending
                            or type(checkpoint_attempt) is not int or checkpoint_attempt!=attempts+1
                            or checkpoint_attempt>len(declared_actions)):
                            _fail()
                    else:
                        _fail()
                if (number!=checkpoint or len(checkpoint_rows)>=len(expected_order)
                    or ident!=expected_order[len(checkpoint_rows)]
                    or row.get('checkpoint_purpose')!=checkpoint_kind
                    or row.get('checkpoint_attempt')!=checkpoint_attempt):
                    _fail()
                required=sorted(d['table_id'] for d in table.get('required_decisions',[]))
                unresolved=sorted(d for d in required if checkpoint_rows.get(d) is None)
                target=row.get('selected_target_activity_id')
                selected_status='first_match' if first_policy else 'unique_match'
                targets=expected_routes[ident]
                expected_match=(target is not None and all(t==target for t in targets))
                if (row.get('required_decision_table_ids')!=required
                    or row.get('unresolved_required_decision_table_ids')!=unresolved
                    or row.get('check_purpose')!=('visited_route' if targets else 'prerequisite_output')
                    or bool(unresolved)!=(row.get('status')=='dependency_unresolved')
                    or (target is not None and (not isinstance(target,str) or target not in {r['target_activity_id'] for r in table['rows']}))
                    or (target is not None)!=(row.get('status')==selected_status)
                    or row['route_matches']!=expected_match or (unresolved and target is not None)):
                    _fail()
                checkpoint_rows[ident]=target
                if len(checkpoint_rows)==len(expected_order):
                    checkpoint_ready=True
                    if checkpoint_kind=='initial':initial_checked=True
                    if checkpoint_kind=='completion':completion_checked=True

            if first_policy:
                if row.get('hit_policy')!='FIRST' or row.get('row_precedence')!='declared_row_order':_fail()
                selected=next((r for r in table['rows'] if r['id']==row.get('selected_row_id')),None)
                if row.get('status')=='first_match':
                    if (selected is None or selected['target_activity_id']!=row.get('selected_target_activity_id')
                        or row.get('blocking_row_id') is not None):_fail()
                elif row.get('selected_row_id') is not None or row.get('selected_target_activity_id') is not None:_fail()
                if row.get('status')!='unknown' and row.get('blocking_row_id') is not None:_fail()
                if row.get('blocking_row_id') is not None and not any(r['id']==row['blocking_row_id'] for r in table['rows']):_fail()
            if row.get('status')!=('first_match' if first_policy else 'unique_match') or not row['route_matches']:
                blocked=True
        if phase == 'catalog_checked' and row.get('current_independent_design_review') is not True:
            blocked = True
        if phase == 'stopped_before_tools':
            if attempts or steps or pending:
                _fail()
            ended = ('handed_to_human', stamp)
        elif phase == 'finished':
            if any(type(row.get(key)) is not bool for key in
                   ('declared_route_checks_satisfied', 'all_tool_results_matched', 'all_declared_operations_verified')):
                _fail()
            succeeded = row['declared_route_checks_satisfied']
            if succeeded and dependent:
                if attempts!=len(declared_actions) or preconditions_seen!=declared_preconditions:
                    _fail()
                if expected_source_rules:
                    required={('initial',None),('completion',None)}|{('action',i+1) for i in range(attempts)}|{('precondition',i) for i in declared_preconditions}
                    if (set(source_checks)!=required or source_key!=('completion',None)
                        or any(rows!=['matched']*len(expected_source_rules) for rows in source_checks.values())):
                        _fail()
                if expected_order and (not initial_checked or not completion_checked or not checkpoint_ready
                    or checkpoint_kind!='completion'):
                    _fail()
            if dependency_required and succeeded and (not dependency_initial or not dependency_completion
                    or attempts!=len(expected_actions) or precondition_attempts!=expected_preconditions):
                _fail()
            if (row['all_tool_results_matched'] and (pending or any(s['status'] != 'completed' for s in steps))
                    or succeeded and (blocked or not row['all_tool_results_matched'])):
                _fail()
            # An explicit uncertain attempt always needs reconciliation, even if
            # another tool reported failure. No result means no invented failure.
            uncertain = pending or any(s['status'] == 'handed_to_human' for s in steps)
            status = 'succeeded' if succeeded else ('failed' if not uncertain and any(
                s['status'] == 'failed' for s in steps) else 'handed_to_human')
            ended = (status, stamp)
        if ended and index != len(records) - 1:
            _fail()
    if pending:
        steps.append({'step_id': pending[0], 'activity': pending[0], 'status': 'handed_to_human',
                      'at': pending[2].isoformat()})
    if len(steps) > 100:
        _fail()
    return {'format': 'openworkgraph.automation-run-report.v1',
            'deployment': {'name': deployment_name.strip(), 'runtime': 'openai-agents',
                           'workflow_spec_sha256': digest(spec)},
            'runs': [{'run_id': first['execution_id'], 'started_at': started.isoformat(),
                      'finished_at': ended[1].isoformat() if ended else None,
                      'status': ended[0] if ended else 'running', 'steps': steps,
                      'outcome': {'verified': None, 'source': 'customer_execution_journal'}}]}


def _review_row(index, content, records, run):
    phases = Counter(row['phase'] for row in records)
    stop = bool(phases['operator_stop_observed'] or phases['operator_stop_active_at_completion'])
    pending = phases['attempt_pending'] - sum(phases[p] for p in (
        'tool_returned_expected_structure', 'tool_reported_failure', 'uncertain'))
    # A precondition request can have been sent without a corresponding
    # business attempt. Do not equate that with a write or a known failure.
    precondition_unresolved = (
        phases['precondition_pending'] > phases['precondition_not_established'] + phases['attempt_pending']
    )
    flags = []
    if run['status'] == 'running':
        flags.append('no_terminal_record')
    if stop:
        flags.append('operator_stop_reported_not_remote_cancellation')
    if pending:
        flags.append('attempt_without_recorded_result')
    if phases['uncertain']:
        flags.append('uncertain_tool_result')
    if precondition_unresolved:
        flags.append('precondition_request_without_recorded_disposition')
    if run['status'] in {'failed', 'handed_to_human'}:
        flags.append('reported_failure_or_handoff')
    return {'selection_index': index, 'journal_sha256': hashlib.sha256(content).hexdigest(),
            'journal_bytes': len(content), 'record_count': len(records),
            'run_id': run['run_id'], 'reported_status': run['status'],
            'started_at': run['started_at'], 'finished_at': run['finished_at'],
            'recorded_attempt_count': phases['attempt_pending'],
            'recorded_uncertain_attempt_count': phases['uncertain'] + pending,
            'operator_stop_reported': stop, 'review_flags': flags,
            'accepted_for_projection': True}


def batch_report(spec, paths, deployment_name):
    """Project explicitly selected journals, with an all-or-nothing report.

    Review rows use 1-based selection indices, not filenames. Each input is
    read once and the exact bytes used for projection are hashed. A corrupt,
    duplicate or over-budget selection never produces a partial run report.
    """
    validate_spec(spec)
    if (not isinstance(paths, (tuple, list)) or not 1 <= len(paths) <= MAX_BATCH_JOURNALS):
        raise ImplementationError('Select 1 to 100 explicit journal files')
    # Validate label independently so it cannot be mistaken for a file failure.
    if (not isinstance(deployment_name, str) or not 1 <= len(deployment_name.strip()) <= 120
            or any(ord(c) < 32 or 127 <= ord(c) <= 159 for c in deployment_name)):
        raise ImplementationError('Use a short deployment label without control characters')
    runs, rows, used, by_id = [], [], 0, {}
    exhausted = False
    for index, path in enumerate(paths, 1):
        if exhausted or used >= MAX_BATCH_BYTES:
            exhausted = True
            rows.append({'selection_index': index, 'accepted_for_projection': False,
                         'refusal': 'batch_byte_budget_exhausted'})
            continue
        limit = min(MAX_BYTES, MAX_BATCH_BYTES - used)
        try:
            content = _read_snapshot(path, limit)
        except (OSError, ValueError, TypeError):
            rows.append({'selection_index': index, 'accepted_for_projection': False,
                         'refusal': 'journal_unreadable'})
            continue
        used += len(content)
        if len(content) > limit:
            exhausted = used > MAX_BATCH_BYTES
            rows.append({'selection_index': index, 'accepted_for_projection': False,
                         'refusal': 'journal_or_batch_byte_limit_exceeded'})
            continue
        try:
            records = _decode_journal(content)
            projected = report(spec, records, deployment_name)
            run = projected['runs'][0]
            row = _review_row(index, content, records, run)
        except (ImplementationError, ValueError, TypeError, KeyError, AttributeError, RecursionError):
            rows.append({'selection_index': index, 'accepted_for_projection': False,
                         'journal_sha256': hashlib.sha256(content).hexdigest(),
                         'journal_bytes': len(content), 'refusal': 'invalid_or_wrong_spec_journal'})
            continue
        rows.append(row)
        runs.append(run)
        by_id.setdefault(run['run_id'], []).append(row)
    for duplicates in by_id.values():
        if len(duplicates) > 1:
            for row in duplicates:
                row['accepted_for_projection'] = False
                row['refusal'] = 'duplicate_execution_id'
    refused = sum(not row['accepted_for_projection'] for row in rows)
    status_counts = Counter(row['reported_status'] for row in rows if row['accepted_for_projection'])
    review = {'format': 'openworkgraph.customer-journal-batch-review.v1',
              'process_reference': reference(spec), 'spec_sha256': digest(spec),
              'selected_journal_count': len(paths), 'accepted_journal_count': len(paths) - refused,
              'refused_journal_count': refused, 'reported_status_counts': dict(sorted(status_counts.items())),
              'journals_requiring_reconciliation': sum(bool(row.get('review_flags')) for row in rows
                                                        if row['accepted_for_projection']),
              'journals': rows, 'complete_selection_projectable': not refused,
              'report_sha256': None, 'export_refusal': None,
              'network_contacted': False, 'business_tools_called': False,
              'business_outcomes_verified': False, 'remote_cancellation_verified': False,
              'execution_authorized': False, 'journal_origin_independently_verified': False}
    if refused:
        review['export_refusal'] = 'one_or_more_selected_journals_refused'
        return None, review
    document = {'format': 'openworkgraph.automation-run-report.v1',
                'deployment': {'name': deployment_name.strip(), 'runtime': 'openai-agents',
                               'workflow_spec_sha256': digest(spec)},
                'runs': sorted(runs, key=lambda run: (run['started_at'], run['run_id']))}
    if len(_encoded(document)) > MAX_EXPORT_BYTES:
        review['complete_selection_projectable'] = False
        review['export_refusal'] = 'signed_report_sender_size_limit_exceeded'
        return None, review
    review['report_sha256'] = digest(document)
    return document, review


def compare_journals(spec, baseline_path, candidate_path, deployment_name):
    """Compare two exact snapshots of one execution without importing either.

    Only an exact byte-prefix extension establishes continuity relative to
    the selected baseline. Equivalent JSON reserialization is not continuity.
    The result never establishes authenticity of either local file.
    """
    validate_spec(spec)
    snapshots = []
    for index, path in enumerate((baseline_path, candidate_path), 1):
        content = _read_snapshot(path)
        records = _decode_journal(content)
        run = report(spec, records, deployment_name)['runs'][0]
        snapshots.append((content, _review_row(index, content, records, run)))
    old, before = snapshots[0]
    new, after = snapshots[1]
    if before['run_id'] != after['run_id']:
        relation = 'different_execution_identity'
    elif old == new:
        relation = 'identical_snapshot'
    elif new.startswith(old):
        relation = 'exact_append_only_extension'
    elif old.startswith(new):
        relation = 'candidate_is_earlier_snapshot'
    else:
        relation = 'divergent_journal_history'
    accepted = relation in {'identical_snapshot', 'exact_append_only_extension'}
    return {
        'format': 'openworkgraph.customer-journal-snapshot-comparison.v1',
        'process_reference': reference(spec), 'spec_sha256': digest(spec),
        'baseline': before, 'candidate': after,
        'relation': relation, 'candidate_retains_exact_baseline_bytes': accepted,
        'additional_record_count': after['record_count'] - before['record_count'] if accepted else None,
        'candidate_requires_reconciliation': bool(after['review_flags']),
        'journal_origin_independently_verified': False,
        'business_outcomes_verified': False, 'remote_cancellation_verified': False,
        'report_imported': False, 'execution_authorized': False,
        'network_contacted': False, 'business_tools_called': False,
    }


def _encoded(document):
    return (json.dumps(document, indent=2, allow_nan=False) + '\n').encode('utf-8')


def _write_new(path, document):
    # Encode before exclusive creation. Never overwrite a selected journal,
    # specification, existing report or other operator file.
    payload = _encoded(document)
    descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(descriptor, 'wb') as stream:
        stream.write(payload)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--journal', required=True, action='append',
                        help='Explicit local journal; repeat for a bounded batch')
    parser.add_argument('--deployment-name', required=True)
    parser.add_argument('--output', required=True, help='New local JSON file; existing files are never overwritten')
    parser.add_argument('--review-only', action='store_true',
                        help='Write a minimized selection/reconciliation review instead of an importable report')
    parser.add_argument('--compare-to', help='Original baseline journal for a read-only comparison with one --journal candidate')
    args = parser.parse_args()
    if args.compare_to is not None and (not args.review_only or len(args.journal) != 1):
        parser.error('--compare-to requires --review-only and exactly one candidate --journal')
    spec = load_object(Path(__file__).parent / 'workflow.spec.json', limit=512 * 1024)
    if args.compare_to is not None:
        review = compare_journals(spec, args.compare_to, args.journal[0], args.deployment_name)
        _write_new(args.output, review)
        return 0 if review['candidate_retains_exact_baseline_bytes'] else 2
    document, review = batch_report(spec, args.journal, args.deployment_name)
    if args.review_only:
        _write_new(args.output, review)
        return 0 if review['complete_selection_projectable'] else 2
    if document is None:
        raise ImplementationError('The complete journal selection cannot be projected; run --review-only')
    _write_new(args.output, document)
    return 0


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except (ImplementationError, OSError, ValueError, KeyError, TypeError, AttributeError, ImportError, RecursionError):
        raise SystemExit('Report not exported. Select a valid exact-spec journal and a new local output file.') from None
