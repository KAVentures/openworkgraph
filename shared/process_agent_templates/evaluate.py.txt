"""Bounded shadow evaluation against explicitly reviewed case expectations.

Offline result comparison is the default. --model explicitly sends selected
case inputs to that provider through the existing no-tools SDK planner.
Neither mode executes business activities or verifies operational outcomes.
"""
import argparse
import asyncio
from collections import Counter
from datetime import datetime, timezone
import json
from pathlib import Path
import re

from execute import ImplementationError, digest, load_object, reference
from runtime import check_plan, validate_spec


def _same_reference(spec, value):
    return isinstance(value, dict) and digest(value) == digest(reference(spec))


def _path(value):
    return (isinstance(value, list) and 1 <= len(value) <= 80
            and all(isinstance(node, str) and 1 <= len(node) <= 64 for node in value))


def validate_suite(spec, suite, *, allow_current_source_rules=False):
    validate_spec(spec)
    if (suite.get('format') != 'openworkgraph.reviewed-shadow-cases.v1'
            or not _same_reference(spec, suite.get('process_reference'))
            or suite.get('spec_sha256') != digest(spec)
            or suite.get('reviewed_by_customer') is not True
            or suite.get('case_inputs_authorized') is not True):
        raise ImplementationError('Use explicitly reviewed, authorized cases for this exact specification')
    cases = suite.get('cases')
    if not isinstance(cases, list) or not 1 <= len(cases) <= 50:
        raise ImplementationError('A shadow suite requires 1 to 50 selected cases')
    seen = set()
    for case in cases:
        if (not isinstance(case, dict) or set(case) != {'id', 'category', 'input', 'expected'}
                or not isinstance(case['id'], str) or not re.fullmatch(r'case_[0-9]{1,4}', case['id'])
                or case['id'] in seen or case['category'] not in ('normal', 'exception', 'missing_facts')
                or not isinstance(case['input'], dict)
                or len(json.dumps(case['input'], allow_nan=False).encode()) > 65536):
            raise ImplementationError('Selected cases require unique opaque IDs, bounded object inputs and declared categories')
        seen.add(case['id'])
        expected = case['expected']
        if (not isinstance(expected, dict) or set(expected) != {'status', 'accepted_routes', 'review_confirmed'}
                or expected['review_confirmed'] is not True
                or expected['status'] not in ('proposal', 'needs_clarification')
                or not isinstance(expected['accepted_routes'], list) or len(expected['accepted_routes']) > 8):
            raise ImplementationError('Every case needs a reviewed route or clarification expectation')
        routes = expected['accepted_routes']
        if expected['status'] == 'needs_clarification':
            if routes:
                raise ImplementationError('Clarification expectations cannot also approve a route')
        elif not routes:
            raise ImplementationError('Route expectations require at least one reviewed path')
        for route in routes:
            if not _path(route):
                raise ImplementationError('Reviewed paths require bounded activity IDs')
            try:
                checks = check_plan({**reference(spec), 'activity_ids': route}, spec)
            except ValueError:
                raise ImplementationError('A reviewed expected route is outside the current process graph') from None
            if checks['unconfirmed_rule_ids'] or checks['manual_review_conditions'] or (checks.get('unevaluated_structured_rule_ids') and not allow_current_source_rules):
                raise ImplementationError('Expected proposals cannot bypass unconfirmed rules or unresolved manual review')
    return cases


def validate_results(spec, suite, results, *, allow_current_source_rules=False):
    cases = validate_suite(spec, suite, allow_current_source_rules=allow_current_source_rules)
    if (results.get('format') != 'openworkgraph.shadow-plan-results.v1'
            or not _same_reference(spec, results.get('process_reference'))
            or results.get('spec_sha256') != digest(spec)
            or results.get('suite_sha256') != digest(suite)
            or not isinstance(results.get('results'), list) or len(results['results']) != len(cases)):
        raise ImplementationError('Select results bound to this exact specification and reviewed suite')
    identifiers = {case['id'] for case in cases}
    found = {}
    for row in results['results']:
        if (not isinstance(row, dict) or set(row) != {'case_id', 'plan'}
                or not isinstance(row['case_id'], str) or row['case_id'] not in identifiers or row['case_id'] in found):
            raise ImplementationError('Results must identify each selected case exactly once')
        found[row['case_id']] = row['plan']
    return found


def compare(spec, suite, results, *, source_rule_contexts=None):
    plans = validate_results(spec, suite, results, allow_current_source_rules=source_rule_contexts is not None)
    rows = []
    for case in suite['cases']:
        plan, reason, checks = plans[case['id']], None, None
        # Malformed or stale individual model output is a failed case, never an
        # instruction or a reason to discard the rest of an offline evaluation.
        if (not isinstance(plan, dict)
                or set(plan) != {'revision', 'content_sha256', 'status', 'activity_ids', 'missing_business_inputs'}
                or type(plan.get('revision')) is not int or plan['revision'] != spec['revision']
                or plan.get('content_sha256') != spec['content_sha256']
                or plan.get('status') not in ('proposal', 'needs_clarification')
                or not isinstance(plan.get('missing_business_inputs'), list) or len(plan['missing_business_inputs']) > 40
                or any(not isinstance(text, str) or not 1 <= len(text) <= 2000 for text in plan['missing_business_inputs'])
                or not isinstance(plan.get('activity_ids'), list) or len(plan['activity_ids']) > 80
                or any(not isinstance(node, str) or not 1 <= len(node) <= 64 for node in plan['activity_ids'])):
            reason = 'invalid_or_stale_plan'
        elif plan['status'] == 'proposal':
            try:
                checks = check_plan(plan, spec)
            except ValueError:
                reason = 'invalid_route'
            unresolved=checks.get('unevaluated_structured_rule_ids',[]) if checks else []
            table_rules={row['rule_id'] for table in spec['process_model'].get('decision_tables',[]) for row in table['rows']}
            unresolved=[rid for rid in unresolved if rid not in table_rules or rid in (checks or {}).get('non_table_structured_rule_ids',[])]
            table_routes=(checks or {}).get('decision_table_routes',[])
            unresolved_tables=bool(table_routes)
            if table_routes and source_rule_contexts is not None:
                context=source_rule_contexts.get(case['id'])
                checked=context.check_tables(table_routes,spec,case['input']) if context is not None else []
                unresolved_tables=len(checked)!=len(table_routes) or any(not row['route_matches'] for row in checked)
            if unresolved and source_rule_contexts is not None:
                context=source_rule_contexts.get(case['id'])
                verified=(context.check_rules(unresolved,spec,case['input']) if context is not None else [])
                unresolved=[rule_id for rule_id in unresolved if not any(
                    row['rule_id']==rule_id and row['status']=='matched' for row in verified)]
            if checks and (checks['unconfirmed_rule_ids'] or checks['manual_review_conditions'] or unresolved or unresolved_tables):
                reason = 'unresolved_review_bypassed'
            if reason is None:
                if plan['missing_business_inputs']:
                    reason = 'proposal_has_unresolved_inputs'
                elif case['expected']['status'] != 'proposal':
                    reason = 'expected_clarification'
                elif plan['activity_ids'] not in case['expected']['accepted_routes']:
                    reason = 'unexpected_route'
        elif not plan['missing_business_inputs']:
            reason = 'clarification_without_question'
        elif any(node not in {row['id'] for row in spec['process_model']['activities']} for node in plan['activity_ids']):
            reason = 'unknown_activity_in_clarification'
        elif case['expected']['status'] != 'needs_clarification':
            reason = 'unexpected_clarification'
        reason = reason or 'matches_reviewed_expectation'
        matched = reason == 'matches_reviewed_expectation'
        rows.append({'case_id': case['id'], 'category': case['category'], 'case_sha256': digest(case['input']),
                     'matches_reviewed_expectation': matched, 'reason': reason,
                     'review_required': not matched})
    return evaluation_report(spec,suite,results,rows)


def evaluation_report(spec,suite,results,rows):
    """Aggregate already checked case rows; no provider claims or raw inputs."""
    counts=Counter(row['reason'] for row in rows)
    return {'format': 'openworkgraph.shadow-evaluation.v1', 'process_reference': reference(spec),
        'spec_sha256': digest(spec), 'suite_sha256': digest(suite), 'results_sha256': digest(results),
        'evaluated_at': datetime.now(timezone.utc).isoformat(), 'case_count': len(rows),
        'matching_case_count': counts['matches_reviewed_expectation'],
        'reason_counts': dict(sorted(counts.items())), 'cases': rows,
        'expectations_are_customer_declarations': True, 'independent_business_ground_truth_verified': False,
        'representative_population_coverage_established': False, 'business_rules_evaluated': False,
        'case_inputs_and_model_text_included': False, 'business_tools_called': False,
        'execution_authorized': False, 'source_outcomes_verified': False,
        'process_model_updated': False, 'review_required': any(row['review_required'] for row in rows)}


def compare_candidates(spec, suite, baseline_results, candidate_results):
    """Paired, offline comparison on the identical reviewed specification and cases.

    Recompute both reports rather than trusting externally supplied scores.
    A matching aggregate does not hide a newly failed case. This is a
    descriptive regression review, not business or deployment authorization.
    """
    baseline = compare(spec, suite, baseline_results)
    candidate = compare(spec, suite, candidate_results)
    old = {row['case_id']: row for row in baseline['cases']}
    rows = []
    counts = Counter()
    by_category = {}
    for new in candidate['cases']:
        before = old[new['case_id']]
        a, b = before['matches_reviewed_expectation'], new['matches_reviewed_expectation']
        change = ('unchanged_match' if a and b else 'regressed' if a
                  else 'improved' if b else 'unchanged_mismatch')
        counts[change] += 1
        group = by_category.setdefault(new['category'], Counter())
        group[change] += 1
        rows.append({
            'case_id': new['case_id'], 'category': new['category'],
            'case_sha256': new['case_sha256'], 'change': change,
            'baseline_reason': before['reason'], 'candidate_reason': new['reason'],
            'candidate_review_required': not b,
        })
    transitions = ('unchanged_match', 'improved', 'regressed', 'unchanged_mismatch')

    def partition(counter):
        return {name: counter[name] for name in transitions}

    categories = {}
    for category in ('normal', 'exception', 'missing_facts'):
        group = by_category.get(category, Counter())
        n = sum(group.values())
        categories[category] = {
            'case_count': n, 'represented': n > 0, **partition(group),
            'baseline_matches': group['unchanged_match'] + group['regressed'],
            'candidate_matches': group['unchanged_match'] + group['improved'],
        }
    # Candidate results may improve overall while newly bypassing an expected
    # clarification or a review condition. Keep these visible separately.
    critical_reasons = ('expected_clarification', 'unresolved_review_bypassed')
    critical = [row for row in rows if row['candidate_reason'] in critical_reasons]
    new_critical = [row for row in critical if row['baseline_reason'] not in critical_reasons]
    return {
        'format': 'openworkgraph.paired-shadow-evaluation.v1',
        'process_reference': reference(spec), 'spec_sha256': digest(spec),
        'suite_sha256': digest(suite),
        'baseline_results_sha256': baseline['results_sha256'],
        'candidate_results_sha256': candidate['results_sha256'],
        'evaluated_at': datetime.now(timezone.utc).isoformat(),
        'case_count': len(rows),
        'baseline_matching_case_count': baseline['matching_case_count'],
        'candidate_matching_case_count': candidate['matching_case_count'],
        'matching_case_count_change': candidate['matching_case_count'] - baseline['matching_case_count'],
        'paired_transitions': partition(counts), 'by_category': categories,
        'candidate_critical_review_failure_count': len(critical),
        'new_critical_review_failure_count': len(new_critical),
        'missing_categories': [name for name, group in categories.items() if not group['represented']],
        'no_new_case_regressions': counts['regressed'] == 0,
        'candidate_all_reviewed_expectations_matched': not candidate['review_required'],
        'review_required': candidate['review_required'] or counts['regressed'] > 0,
        'cases': rows,
        'expectations_are_customer_declarations': True,
        'independent_business_ground_truth_verified': False,
        'representative_population_coverage_established': False,
        'case_inputs_and_model_text_included': False,
        'business_tools_called': False, 'execution_authorized': False,
        'source_outcomes_verified': False, 'process_model_updated': False,
        'model_identity_independently_verified': False,
    }


def _repeatability_summary(reports):
    """Summarize repeated matched-case scores without assuming independence."""
    if not 2 <= len(reports) <= 10:
        raise ImplementationError('Select 2 to 10 captured result documents')
    first = reports[0]
    basis = ('process_reference', 'spec_sha256', 'suite_sha256')
    indexed = []
    for report in reports:
        if any(report[key] != first[key] for key in basis):
            raise ImplementationError('Repeated review requires an identical specification and suite')
        rows = {row['case_id']: row for row in report['cases']}
        if len(rows) != len(report['cases']) or not rows:
            raise ImplementationError('Repeated review requires unique selected cases')
        indexed.append(rows)
    ids = set(indexed[0])
    if any(set(rows) != ids for rows in indexed):
        raise ImplementationError('Repeated review requires identical case coverage')
    critical = {'expected_clarification', 'unresolved_review_bypassed'}
    result = []
    for case_id in sorted(ids):
        rows = [capture[case_id] for capture in indexed]
        base = rows[0]
        if any((r['case_sha256'], r['category']) != (base['case_sha256'], base['category']) for r in rows):
            raise ImplementationError('Repeated review requires identical case identity and category')
        matches = sum(r['matches_reviewed_expectation'] for r in rows)
        n = len(rows)
        result.append({
            'case_id': case_id, 'case_sha256': base['case_sha256'], 'category': base['category'],
            'matching_captures': matches, 'mismatching_captures': n - matches,
            'status': 'always_matches' if matches == n else 'never_matches' if not matches else 'intermittent_match',
            'critical_review_failure_captures': sum(r['reason'] in critical for r in rows),
            'reason_counts': dict(sorted(Counter(r['reason'] for r in rows).items())),
            'capture_results': [{'capture_index': i + 1, 'reason': r['reason'],
                                 'matches_reviewed_expectation': r['matches_reviewed_expectation']}
                                for i, r in enumerate(rows)],
        })
    def summarize(rows):
        counts = Counter(r['status'] for r in rows)
        return {'case_count': len(rows), 'always_matches': counts['always_matches'],
                'intermittent_match': counts['intermittent_match'], 'never_matches': counts['never_matches'],
                'cases_with_any_critical_failure': sum(r['critical_review_failure_captures'] > 0 for r in rows)}
    totals = [sum(r['matches_reviewed_expectation'] for r in capture.values()) for capture in indexed]
    hashes = Counter(r['results_sha256'] for r in reports)
    return {
        'capture_count': len(reports), 'distinct_result_document_count': len(hashes),
        'duplicate_result_document_occurrences': len(reports) - len(hashes),
        'captures': [{'capture_index': i + 1, 'results_sha256': r['results_sha256'],
                      'matching_case_count': totals[i]} for i, r in enumerate(reports)],
        'minimum_matching_case_count': min(totals), 'maximum_matching_case_count': max(totals),
        'mean_matching_case_fraction': round(sum(totals) / (len(ids) * len(reports)), 6),
        'case_summary': summarize(result), 'by_category': {
            category: summarize([r for r in result if r['category'] == category])
            for category in ('normal', 'exception', 'missing_facts')},
        'cases': result,
        'review_required': any(r['status'] != 'always_matches' for r in result),
    }


def compare_repeated_results(spec, suite, captures):
    """Recompute every offline result using the same reviewed case contract."""
    if not isinstance(captures, (list, tuple)) or not 2 <= len(captures) <= 10:
        raise ImplementationError('Select 2 to 10 captured result documents')
    reports = [compare(spec, suite, capture) for capture in captures]
    summary = _repeatability_summary(reports)
    return {'format': 'openworkgraph.repeated-shadow-evaluation.v1',
            'process_reference': reference(spec), 'spec_sha256': digest(spec), 'suite_sha256': digest(suite),
            'evaluated_at': datetime.now(timezone.utc).isoformat(), **summary,
            'case_count': summary['case_summary']['case_count'],
            'missing_categories': [key for key, row in summary['by_category'].items() if not row['case_count']],
            'expectations_are_customer_declarations': True,
            'capture_independence_verified': False, 'same_model_or_configuration_verified': False,
            'statistical_confidence_interval': False, 'model_identity_independently_verified': False,
            'representative_population_coverage_established': False, 'independent_business_ground_truth_verified': False,
            'case_inputs_and_model_text_included': False, 'business_tools_called': False,
            'execution_authorized': False, 'source_outcomes_verified': False, 'process_model_updated': False}


async def run_planner(spec, suite, *, model):
    cases = validate_suite(spec, suite)
    from agent import propose
    # Expectations are deliberately excluded: the planner sees only each case
    # input and the selected model, using the existing tracing-disabled agent.
    rows = []
    for case in cases:
        result = await propose(case['input'], model=model, spec=spec)
        rows.append({'case_id': case['id'], 'plan': result['plan']})
    return {'format': 'openworkgraph.shadow-plan-results.v1', 'process_reference': reference(spec),
            'spec_sha256': digest(spec), 'suite_sha256': digest(suite), 'results': rows}


def main():
    parser = argparse.ArgumentParser(description='Compare reviewed shadow cases; no business actions or automatic process updates')
    parser.add_argument('--cases', required=True, help='Explicitly authorized case inputs and reviewed expectations')
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument('--results', help='Offline previously captured plans bound to the exact suite')
    mode.add_argument('--model', help='Explicit provider opt-in: sends case inputs through the no-tools planner')
    parser.add_argument('--baseline-results', help='Optional offline baseline plans for the identical reviewed suite; requires --results')
    parser.add_argument('--repeat-results', action='append', help='Additional offline result document; repeat up to nine times with --results')
    parser.add_argument('--capture-results', help='Explicitly retain live model plans in a new private file for later paired evaluation; may include sensitive model text')
    parser.add_argument('--confirm-retain-model-output', action='store_true', help='Confirm local retention of possibly sensitive model output')
    parser.add_argument('--output', required=True, help='New minimized report file; never overwrites existing files')
    args = parser.parse_args()
    if args.repeat_results and (not args.results or args.baseline_results or len(args.repeat_results) > 9):
        parser.error('--repeat-results requires offline --results, at most nine additions and no --baseline-results')
    if args.baseline_results and not args.results:
        parser.error('--baseline-results requires offline --results')
    if args.capture_results and (not args.model or not args.confirm_retain_model_output):
        parser.error('--capture-results requires --model and --confirm-retain-model-output')
    if args.confirm_retain_model_output and not args.capture_results:
        parser.error('--confirm-retain-model-output requires --capture-results')
    if Path(args.output).exists():
        raise ImplementationError('Choose a new evaluation report file')
    if args.capture_results and (Path(args.capture_results).exists() or
            Path(args.capture_results).resolve() == Path(args.output).resolve()):
        raise ImplementationError('Choose a separate new captured-results file')
    spec = load_object(Path(__file__).parent / 'workflow.spec.json', limit=262144)
    suite = load_object(args.cases, limit=262144)
    validate_suite(spec, suite)
    results = (asyncio.run(run_planner(spec, suite, model=args.model)) if args.model
               else load_object(args.results, limit=262144))
    if args.repeat_results:
        report = compare_repeated_results(spec, suite, [results] + [
            load_object(path, limit=262144) for path in args.repeat_results])
    else:
        report = (compare_candidates(spec, suite, load_object(args.baseline_results, limit=262144), results)
                  if args.baseline_results else compare(spec, suite, results))
    if args.capture_results:
        # These are raw planner outputs, unlike the minimized evaluation report.
        # Exclusive creation and a private mode on POSIX; Windows inherits ACLs.
        import os
        payload = json.dumps(results, separators=(',', ':'), allow_nan=False).encode('utf-8')
        if len(payload) > 262144:
            raise ImplementationError('Captured plans exceed the supported offline results size; select fewer cases')
        fd = os.open(args.capture_results, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        with os.fdopen(fd, 'wb') as stream:
            stream.write(payload)
    with Path(args.output).open('x', encoding='utf-8') as stream:
        json.dump(report, stream, indent=2, allow_nan=False)
    summary = {'case_count': report['case_count'], 'review_required': report['review_required'],
               'business_tools_called': False}
    if args.repeat_results:
        summary.update(capture_count=report['capture_count'],case_summary=report['case_summary'])
    elif args.baseline_results:
        summary.update(baseline_matching_case_count=report['baseline_matching_case_count'],
                       candidate_matching_case_count=report['candidate_matching_case_count'],
                       paired_transitions=report['paired_transitions'])
    else:
        summary['matching_case_count'] = report['matching_case_count']
    print(json.dumps(summary))
    return 2 if report['review_required'] else 0


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except ImplementationError as exc:
        raise SystemExit(str(exc)) from None
    except Exception:
        raise SystemExit('Shadow evaluation stopped; review exact specification, suite and selected model/results. No execution was approved.') from None
