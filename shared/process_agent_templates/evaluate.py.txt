"""Bounded shadow evaluation against explicitly reviewed case expectations.

Offline result comparison is the default. --model explicitly sends selected
case inputs to that provider through the existing no-tools SDK planner.
Neither mode executes business activities or verifies operational outcomes.
"""
import argparse
import asyncio
from collections import Counter
from datetime import datetime, timezone
import json
from pathlib import Path
import re

from execute import ImplementationError, digest, load_object, reference
from runtime import check_plan, validate_spec


def _same_reference(spec, value):
    return isinstance(value, dict) and digest(value) == digest(reference(spec))


def _path(value):
    return (isinstance(value, list) and 1 <= len(value) <= 80
            and all(isinstance(node, str) and 1 <= len(node) <= 64 for node in value))


def validate_suite(spec, suite, *, allow_current_source_rules=False):
    validate_spec(spec)
    if (suite.get('format') != 'openworkgraph.reviewed-shadow-cases.v1'
            or not _same_reference(spec, suite.get('process_reference'))
            or suite.get('spec_sha256') != digest(spec)
            or suite.get('reviewed_by_customer') is not True
            or suite.get('case_inputs_authorized') is not True):
        raise ImplementationError('Use explicitly reviewed, authorized cases for this exact specification')
    cases = suite.get('cases')
    if not isinstance(cases, list) or not 1 <= len(cases) <= 50:
        raise ImplementationError('A shadow suite requires 1 to 50 selected cases')
    seen = set()
    for case in cases:
        if (not isinstance(case, dict) or set(case) != {'id', 'category', 'input', 'expected'}
                or not isinstance(case['id'], str) or not re.fullmatch(r'case_[0-9]{1,4}', case['id'])
                or case['id'] in seen or case['category'] not in ('normal', 'exception', 'missing_facts')
                or not isinstance(case['input'], dict)
                or len(json.dumps(case['input'], allow_nan=False).encode()) > 65536):
            raise ImplementationError('Selected cases require unique opaque IDs, bounded object inputs and declared categories')
        seen.add(case['id'])
        expected = case['expected']
        if (not isinstance(expected, dict) or set(expected) != {'status', 'accepted_routes', 'review_confirmed'}
                or expected['review_confirmed'] is not True
                or expected['status'] not in ('proposal', 'needs_clarification')
                or not isinstance(expected['accepted_routes'], list) or len(expected['accepted_routes']) > 8):
            raise ImplementationError('Every case needs a reviewed route or clarification expectation')
        routes = expected['accepted_routes']
        if expected['status'] == 'needs_clarification':
            if routes:
                raise ImplementationError('Clarification expectations cannot also approve a route')
        elif not routes:
            raise ImplementationError('Route expectations require at least one reviewed path')
        for route in routes:
            if not _path(route):
                raise ImplementationError('Reviewed paths require bounded activity IDs')
            try:
                checks = check_plan({**reference(spec), 'activity_ids': route}, spec)
            except ValueError:
                raise ImplementationError('A reviewed expected route is outside the current process graph') from None
            if checks['unconfirmed_rule_ids'] or checks['manual_review_conditions'] or (checks.get('unevaluated_structured_rule_ids') and not allow_current_source_rules):
                raise ImplementationError('Expected proposals cannot bypass unconfirmed rules or unresolved manual review')
    return cases


def validate_results(spec, suite, results, *, allow_current_source_rules=False):
    cases = validate_suite(spec, suite, allow_current_source_rules=allow_current_source_rules)
    if (results.get('format') != 'openworkgraph.shadow-plan-results.v1'
            or not _same_reference(spec, results.get('process_reference'))
            or results.get('spec_sha256') != digest(spec)
            or results.get('suite_sha256') != digest(suite)
            or not isinstance(results.get('results'), list) or len(results['results']) != len(cases)):
        raise ImplementationError('Select results bound to this exact specification and reviewed suite')
    identifiers = {case['id'] for case in cases}
    found = {}
    for row in results['results']:
        if (not isinstance(row, dict) or set(row) != {'case_id', 'plan'}
                or not isinstance(row['case_id'], str) or row['case_id'] not in identifiers or row['case_id'] in found):
            raise ImplementationError('Results must identify each selected case exactly once')
        found[row['case_id']] = row['plan']
    return found


def compare(spec, suite, results, *, source_rule_contexts=None):
    plans = validate_results(spec, suite, results, allow_current_source_rules=source_rule_contexts is not None)
    rows = []
    for case in suite['cases']:
        plan, reason, checks = plans[case['id']], None, None
        # Malformed or stale individual model output is a failed case, never an
        # instruction or a reason to discard the rest of an offline evaluation.
        if (not isinstance(plan, dict)
                or set(plan) != {'revision', 'content_sha256', 'status', 'activity_ids', 'missing_business_inputs'}
                or type(plan.get('revision')) is not int or plan['revision'] != spec['revision']
                or plan.get('content_sha256') != spec['content_sha256']
                or plan.get('status') not in ('proposal', 'needs_clarification')
                or not isinstance(plan.get('missing_business_inputs'), list) or len(plan['missing_business_inputs']) > 40
                or any(not isinstance(text, str) or not 1 <= len(text) <= 2000 for text in plan['missing_business_inputs'])
                or not isinstance(plan.get('activity_ids'), list) or len(plan['activity_ids']) > 80
                or any(not isinstance(node, str) or not 1 <= len(node) <= 64 for node in plan['activity_ids'])):
            reason = 'invalid_or_stale_plan'
        elif plan['status'] == 'proposal':
            try:
                checks = check_plan(plan, spec)
            except ValueError:
                reason = 'invalid_route'
            unresolved=checks.get('unevaluated_structured_rule_ids',[]) if checks else []
            table_rules={row['rule_id'] for table in spec['process_model'].get('decision_tables',[]) for row in table['rows']}
            unresolved=[rid for rid in unresolved if rid not in table_rules or rid in (checks or {}).get('non_table_structured_rule_ids',[])]
            table_routes=(checks or {}).get('decision_table_routes',[])
            unresolved_tables=bool(table_routes)
            if table_routes and source_rule_contexts is not None:
                context=source_rule_contexts.get(case['id'])
                checked=context.check_tables(table_routes,spec,case['input']) if context is not None else []
                unresolved_tables=len(checked)!=len(table_routes) or any(not row['route_matches'] for row in checked)
            if unresolved and source_rule_contexts is not None:
                context=source_rule_contexts.get(case['id'])
                verified=(context.check_rules(unresolved,spec,case['input']) if context is not None else [])
                unresolved=[rule_id for rule_id in unresolved if not any(
                    row['rule_id']==rule_id and row['status']=='matched' for row in verified)]
            if checks and (checks['unconfirmed_rule_ids'] or checks['manual_review_conditions'] or unresolved or unresolved_tables):
                reason = 'unresolved_review_bypassed'
            if reason is None:
                if plan['missing_business_inputs']:
                    reason = 'proposal_has_unresolved_inputs'
                elif case['expected']['status'] != 'proposal':
                    reason = 'expected_clarification'
                elif plan['activity_ids'] not in case['expected']['accepted_routes']:
                    reason = 'unexpected_route'
        elif not plan['missing_business_inputs']:
            reason = 'clarification_without_question'
        elif any(node not in {row['id'] for row in spec['process_model']['activities']} for node in plan['activity_ids']):
            reason = 'unknown_activity_in_clarification'
        elif case['expected']['status'] != 'needs_clarification':
            reason = 'unexpected_clarification'
        reason = reason or 'matches_reviewed_expectation'
        matched = reason == 'matches_reviewed_expectation'
        rows.append({'case_id': case['id'], 'category': case['category'], 'case_sha256': digest(case['input']),
                     'matches_reviewed_expectation': matched, 'reason': reason,
                     'review_required': not matched})
    return evaluation_report(spec,suite,results,rows)


def evaluation_report(spec,suite,results,rows):
    """Aggregate already checked case rows; no provider claims or raw inputs."""
    counts=Counter(row['reason'] for row in rows)
    return {'format': 'openworkgraph.shadow-evaluation.v1', 'process_reference': reference(spec),
        'spec_sha256': digest(spec), 'suite_sha256': digest(suite), 'results_sha256': digest(results),
        'evaluated_at': datetime.now(timezone.utc).isoformat(), 'case_count': len(rows),
        'matching_case_count': counts['matches_reviewed_expectation'],
        'reason_counts': dict(sorted(counts.items())), 'cases': rows,
        'expectations_are_customer_declarations': True, 'independent_business_ground_truth_verified': False,
        'representative_population_coverage_established': False, 'business_rules_evaluated': False,
        'case_inputs_and_model_text_included': False, 'business_tools_called': False,
        'execution_authorized': False, 'source_outcomes_verified': False,
        'process_model_updated': False, 'review_required': any(row['review_required'] for row in rows)}


async def run_planner(spec, suite, *, model):
    cases = validate_suite(spec, suite)
    from agent import propose
    # Expectations are deliberately excluded: the planner sees only each case
    # input and the selected model, using the existing tracing-disabled agent.
    rows = []
    for case in cases:
        result = await propose(case['input'], model=model, spec=spec)
        rows.append({'case_id': case['id'], 'plan': result['plan']})
    return {'format': 'openworkgraph.shadow-plan-results.v1', 'process_reference': reference(spec),
            'spec_sha256': digest(spec), 'suite_sha256': digest(suite), 'results': rows}


def main():
    parser = argparse.ArgumentParser(description='Compare reviewed shadow cases; no business actions or automatic process updates')
    parser.add_argument('--cases', required=True, help='Explicitly authorized case inputs and reviewed expectations')
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument('--results', help='Offline previously captured plans bound to the exact suite')
    mode.add_argument('--model', help='Explicit provider opt-in: sends case inputs through the no-tools planner')
    parser.add_argument('--output', required=True, help='New minimized report file; never overwrites existing files')
    args = parser.parse_args()
    if Path(args.output).exists():
        raise ImplementationError('Choose a new evaluation report file')
    spec = load_object(Path(__file__).parent / 'workflow.spec.json', limit=262144)
    suite = load_object(args.cases, limit=262144)
    validate_suite(spec, suite)
    results = (asyncio.run(run_planner(spec, suite, model=args.model)) if args.model
               else load_object(args.results, limit=262144))
    report = compare(spec, suite, results)
    with Path(args.output).open('x', encoding='utf-8') as stream:
        json.dump(report, stream, indent=2, allow_nan=False)
    print(json.dumps({'case_count': report['case_count'], 'matching_case_count': report['matching_case_count'],
                      'review_required': report['review_required'], 'business_tools_called': False}))
    return 2 if report['review_required'] else 0


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except ImplementationError as exc:
        raise SystemExit(str(exc)) from None
    except Exception:
        raise SystemExit('Shadow evaluation stopped; review exact specification, suite and selected model/results. No execution was approved.') from None
