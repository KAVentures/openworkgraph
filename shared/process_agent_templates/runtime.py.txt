"""Offline structural checks. Never executes process activities or connector calls."""
import json
import hashlib
from pathlib import Path


def load_spec():
    spec = json.loads((Path(__file__).parent / 'workflow.spec.json').read_text(encoding='utf-8'))
    validate_spec(spec)
    return spec


def validate_spec(spec):
    tables = spec['process_model'].get('decision_tables')
    if tables:
        if spec.get('decision_table_contract') != 'openworkgraph.current-source-unique-decision.v1':
            raise ValueError('Decision tables require the exported current-source UNIQUE contract')
        from owg_model.process_model import ProcessModel
        ProcessModel.model_validate(spec['process_model'])
    parallel = any(node.get('kind') in {'parallel_split','parallel_join'} for node in spec['process_model']['activities'])
    if parallel:
        if (spec.get('control_flow_contract') != 'openworkgraph.parallel-node-firing-plan.v1'
                or spec.get('execution_schedule') != 'serial_token_valid_node_order'):
            raise ValueError('This linear runtime cannot plan or execute parallel fork/join designs without an exported token-flow contract')
        from owg_model.process_model import ProcessModel
        # Use the same complete graph validator as OWG, including every join.
        ProcessModel.model_validate(spec['process_model'])
    digest = hashlib.sha256(json.dumps(spec['process_model'], sort_keys=True).encode()).hexdigest()
    if (digest != spec['process_model_sha256'] or spec.get('execution_authorization') is not False
            or spec.get('live_business_tools') != []):
        raise ValueError('Process model changed; export a reviewed revision before planning')


def check_plan(plan, spec=None):
    spec = spec or load_spec()
    validate_spec(spec)
    model = spec['process_model']
    ids = plan.get('activity_ids')
    if (type(plan.get('revision')) is not int or plan['revision'] != spec['revision']
            or plan.get('content_sha256') != spec['content_sha256']):
        raise ValueError('Plan does not identify this exact reviewed process revision')
    if (not isinstance(ids, list) or not 1 <= len(ids) <= 80
            or any(not isinstance(node, str) for node in ids)):
        raise ValueError('Plan requires a bounded sequence of activity IDs')
    nodes = {node['id']: node for node in model['activities']}
    edges = {(edge['source'], edge['target']): edge for edge in model['transitions']}
    if (any(node not in nodes for node in ids) or ids[0] != model['entry_activity_id']
            or ids[-1] not in model['terminal_activity_ids']):
        raise ValueError('Proposed path does not fit the reviewed process graph')
    token_checks = None
    if spec.get('control_flow_contract') == 'openworkgraph.parallel-node-firing-plan.v1':
        from owg_model.process_model import ProcessModel
        from owg_model.process_parallel import replay_node_ids
        token_checks = replay_node_ids(ids, ProcessModel.model_validate(model))
        used_rules = set(token_checks['rule_ids'])
    else:
        if any(pair not in edges for pair in zip(ids, ids[1:])):
            raise ValueError('Proposed path does not fit the reviewed process graph')
        used_rules = {edges[pair]['rule_id'] for pair in zip(ids, ids[1:]) if edges[pair].get('rule_id')}
    unconfirmed = sorted(rule['id'] for rule in model['business_rules']
                         if rule['id'] in used_rules and not rule['user_confirmed'])
    checks = {
        'mode': 'offline_structural_shadow_check',
        'workflow_id': spec['workflow_id'], 'revision': spec['revision'],
        'content_sha256': spec['content_sha256'],
        'valid_structure': True,
        'unconfirmed_rule_ids': unconfirmed,
        'requires_tool_mapping_ids': sorted({node for node in ids if nodes[node]['kind'] == 'activity'
                                            and nodes[node]['implementation'] != 'human'}),
        'human_activity_ids': sorted({node for node in ids if nodes[node]['implementation'] == 'human'}),
        'manual_review_conditions': [condition for condition in model.get('manual_review_conditions', [])
                                     if condition['activity_id'] in ids],
        'manual_review_conditions_evaluated': False,
        'business_rules_evaluated': False,
        'execution_permitted': False,
        'source_outcome_verified': False,
    }
    structured = sorted(rule['id'] for rule in model['business_rules']
                        if rule['id'] in used_rules and rule.get('predicate') is not None)
    if structured:
        # A declaration/data comparison cannot attest fresh source facts or
        # become permission for the customer executor to choose a branch.
        checks['unevaluated_structured_rule_ids'] = structured
    if model.get('decision_tables'):
        used_edges = ([model['transitions'][i] for i in token_checks['transition_indices']] if token_checks is not None
                      else [edges[pair] for pair in zip(ids, ids[1:])])
        table_decisions={table['decision_activity_id'] for table in model['decision_tables']}
        checks['non_table_structured_rule_ids']=sorted({edge['rule_id'] for edge in used_edges
            if edge.get('rule_id') and edge['source'] not in table_decisions
            and next(r for r in model['business_rules'] if r['id']==edge['rule_id']).get('predicate') is not None})
        checks['decision_table_routes'] = [
            {'table_id': table['id'], 'target_activity_ids': [edge['target'] for edge in used_edges
                 if edge['source'] == table['decision_activity_id']]}
            for table in model['decision_tables'] if any(edge['source'] == table['decision_activity_id'] for edge in used_edges)]
    if token_checks is not None:
        checks.update(control_flow_contract=spec['control_flow_contract'],
                      execution_schedule=spec['execution_schedule'],
                      all_produced_branch_tokens_consumed=True,
                      token_replay_work_count=token_checks['work_count'])
    return checks
