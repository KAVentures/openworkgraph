def documents(origin="https://customer.example"):
    spec,plan,case,bindings,_=fixture()
    doc={"format":FORMAT,"process_reference":bindings["process_reference"],"spec_sha256":digest(spec),
         "origin":origin,"reviewed":True,"allow_writes":True,"operations":[
        {"name":"read_case","method":"GET","path":"/records/{issue}","fields":{"issue":"string"},
         "query":{},"body":{},"expected_version_argument":None,"idempotency_argument":None,
         "reviewed_provider_contract":True},
        {"name":"resolve_case","method":"PATCH","path":"/records/{issue}","fields":{
            "issue":"string","expected_version":"integer","request_id":"string"},
         "query":{},"body":{},"expected_version_argument":"expected_version",
         "idempotency_argument":"request_id","reviewed_provider_contract":True}]}
    ops=Operations(spec,doc,digest(doc))
    bindings["tool_server_url"]=origin
    for binding in bindings["bindings"]:
        binding["tool_schema_sha256"]=digest(ops.schemas[binding["tool_name"]])
        if binding.get("precondition"):
            condition=binding["precondition"]
            condition["tool_schema_sha256"]=digest(ops.schemas[condition["tool_name"]])
    intent=review_template(spec,prepare(spec,plan,case,bindings),case,bindings)
    intent.update(customer_authorization_confirmed=True,source_reads_confirmed=True,
                  expires_at=(datetime.now(timezone.utc)+timedelta(minutes=10)).isoformat())
    return spec,plan,case,bindings,intent,doc,ops


@unittest.skipUnless(HTTP_AVAILABLE,"Install requirements.txt for direct HTTP runtime tests")
class HttpRuntime(unittest.IsolatedAsyncioTestCase):
    async def asyncSetUp(self):
        self.temp=TemporaryDirectory()
        self.addCleanup(self.temp.cleanup)
        self.environment=patch.dict("os.environ",{"OWG_EXECUTION_HTTP_TOKEN":"PRIVATE_HTTP_TOKEN",
            "OWG_EXECUTION_SOURCE_READ_TOKEN":"PRIVATE_SOURCE_TOKEN","OWG_EXECUTION_MCP_TOKEN":"UNUSED_MCP_TOKEN"})
        self.environment.start()
        self.addCleanup(self.environment.stop)
        self.calls=[]
        self.resolved=False
        self.version=3
        self.failure=None
        self.data=documents()

    async def source(self,base,token,request):
        self.assertEqual(token,"PRIVATE_SOURCE_TOKEN")
        value=receipt(request)
        if not self.resolved:
            value.update(operation_verified=False,availability="unknown")
        return value

    def transport(self,request):
        self.calls.append(request)
        self.assertEqual(request.headers["Authorization"],"Bearer PRIVATE_HTTP_TOKEN")
        self.assertEqual(request.url.host,"customer.example")
        self.assertEqual(request.url.path,"/records/SYNTHETIC_PRIVATE_RECORD")
        if request.method=="PATCH":
            self.assertEqual(request.headers["If-Match"],'"3"')
            self.assertEqual(request.headers["Idempotency-Key"],"synthetic-idempotency-key")
            self.assertEqual(json.loads(request.content),{})
            if self.failure=="redirect":
                return httpx.Response(307,headers={"Location":"https://other.invalid"})
            if self.failure=="timeout":
                raise httpx.ReadTimeout("PRIVATE_TRANSPORT_ERROR")
            if self.failure=="bad_json":
                return httpx.Response(200,headers={"Content-Type":"application/json"},content=b'{"success":true,"success":false}')
            if self.failure=="oversized":
                return httpx.Response(200,json={"success":True,"data":"x"*70000})
            self.resolved=True
        return httpx.Response(200,json={"version":self.version,"success":True,"private":"PRIVATE_PROVIDER_RESPONSE"})

    def factory(self,**kwargs):
        self.assertIs(kwargs["follow_redirects"],False)
        self.assertIs(kwargs["trust_env"],False)
        return httpx.AsyncClient(transport=httpx.MockTransport(self.transport),**kwargs)

    async def execute(self,data=None,folder="journal"):
        spec,plan,case,bindings,intent,doc,ops=data or self.data
        return await run(spec,plan,case,bindings,intent,ops,journal_directory=Path(self.temp.name)/folder,
                         client_factory=self.factory,receipt_reader=self.source)

    async def test_complete_real_executor_and_durable_no_replay(self):
        report=await self.execute()
        self.assertEqual([x.method for x in self.calls],["GET","PATCH","GET"])
        self.assertTrue(report["all_tool_results_matched"])
        self.assertTrue(report["all_declared_operations_verified"])
        self.assertFalse(report["business_objective_verified"])
        journal="".join(p.read_text() for p in (Path(self.temp.name)/"journal").glob("*.jsonl"))
        for private in ("PRIVATE_HTTP_TOKEN","PRIVATE_SOURCE_TOKEN","SYNTHETIC_PRIVATE_RECORD","PRIVATE_PROVIDER_RESPONSE"):
            self.assertNotIn(private,journal)
        with self.assertRaises(ImplementationError):
            await self.execute()
        self.assertEqual(len(self.calls),3)

    async def test_failed_precondition_stops_write(self):
        self.version=4
        report=await self.execute()
        self.assertEqual([x.method for x in self.calls],["GET"])
        self.assertFalse(report["all_tool_results_matched"])
        self.assertFalse(self.resolved)

    async def test_uncertain_write_outcomes_are_not_retried_or_followed(self):
        for failure in ("redirect","timeout","bad_json","oversized"):
            self.failure=failure
            self.calls.clear()
            report=await self.execute(folder=failure)
            self.assertEqual([x.method for x in self.calls],["GET","PATCH"])
            self.assertFalse(report["all_tool_results_matched"])
            self.assertFalse(report["all_declared_operations_verified"])

    def repin(self,data):
        spec,plan,case,bindings,intent,doc,ops=data
        intent=review_template(spec,prepare(spec,plan,case,bindings),case,bindings)
        intent.update(customer_authorization_confirmed=True,source_reads_confirmed=True,
                      expires_at=(datetime.now(timezone.utc)+timedelta(minutes=10)).isoformat())
        return spec,plan,case,bindings,intent,doc,ops

    async def test_unpinned_catalog_or_wrong_method_cannot_call_provider(self):
        for mutation in ("pin","kind","header","origin","path"):
            data=list(deepcopy(self.data))
            spec,plan,case,bindings,intent,doc,ops=data
            if mutation=="pin":
                bindings["bindings"][0]["tool_schema_sha256"]=None
            elif mutation=="kind":
                bindings["bindings"][1]["tool_name"]="resolve_case"
                bindings["bindings"][1]["tool_schema_sha256"]=digest(ops.schemas["resolve_case"])
            elif mutation=="header":
                bindings["bindings"][0]["idempotency"]["argument"]="issue"
            elif mutation=="origin":
                bindings["tool_server_url"]="https://other.invalid"
            elif mutation=="path":
                case["issue"]="../private"
            with self.assertRaises(ImplementationError):
                await self.execute(self.repin(data),folder=mutation)
        self.assertEqual(self.calls,[])

    async def test_actual_loopback_http_write_and_readback(self):
        state={"version":3,"resolved":False,"calls":[]}
        class Handler(BaseHTTPRequestHandler):
            def log_message(self,*args):
                pass
            def respond(self,status,payload):
                raw=json.dumps(payload).encode()
                self.send_response(status)
                self.send_header("Content-Type","application/json")
                self.send_header("Content-Length",str(len(raw)))
                self.end_headers()
                self.wfile.write(raw)
            def do_GET(self):
                state["calls"].append(self.command)
                if self.headers.get("Authorization")!="Bearer PRIVATE_HTTP_TOKEN":
                    return self.respond(403,{"success":False})
                self.respond(200,{"success":True,"version":state["version"]})
            def do_PATCH(self):
                state["calls"].append(self.command)
                self.rfile.read(int(self.headers.get("Content-Length","0")))
                if (self.headers.get("Authorization")!="Bearer PRIVATE_HTTP_TOKEN"
                    or self.headers.get("If-Match")!='"3"'
                    or self.headers.get("Idempotency-Key")!="synthetic-idempotency-key"):
                    return self.respond(412,{"success":False})
                state["resolved"]=True
                state["version"]=4
                self.respond(200,{"success":True,"version":4})
        server=ThreadingHTTPServer(("127.0.0.1",0),Handler)
        thread=threading.Thread(target=server.serve_forever,daemon=True)
        thread.start()
        try:
            spec,plan,case,bindings,intent,doc,ops=documents("http://127.0.0.1:"+str(server.server_port))
            async def source(base,token,request):
                self.assertTrue(state["resolved"])
                return receipt(request)
            report=await run(spec,plan,case,bindings,intent,ops,
                journal_directory=Path(self.temp.name)/"real-http",receipt_reader=source)
            self.assertEqual(state["calls"],["GET","PATCH","GET"])
            self.assertTrue(report["all_tool_results_matched"])
            self.assertTrue(report["all_declared_operations_verified"])
            self.assertEqual(state["version"],4)
        finally:
            server.shutdown()
            server.server_close()
            thread.join(timeout=3)

    async def test_exact_offline_http_case_suite_rehearsal(self):
        from rehearse_http import review
        spec,plan,case,bindings,_,doc,ops=self.data
        suite={"format":"openworkgraph.reviewed-shadow-cases.v1",
               "process_reference":bindings["process_reference"],"spec_sha256":digest(spec),
               "reviewed_by_customer":True,"case_inputs_authorized":True,
               "cases":[{"id":"case_1","category":"normal","input":deepcopy(case),
                         "expected":{"status":"proposal","accepted_routes":[plan["activity_ids"]],
                                     "review_confirmed":True}}]}
        with patch("execute_http.DirectClient",side_effect=AssertionError("Unexpected transport")):
            good=review(spec,suite,bindings,doc,digest(doc))
            self.assertTrue(good["all_cases_match_offline"])
            self.assertEqual(good["counts"]["matching_routes"],1)
            self.assertEqual(good["cases"][0]["routes"][0]["counts"]["terminal_source_checks"],1)
            self.assertFalse(good["execution_authorized"])
            self.assertEqual(good["review_sha256"],digest({k:v for k,v in good.items() if k!="review_sha256"}))
            for private in ("SYNTHETIC_PRIVATE_RECORD","synthetic-idempotency-key",case["record_ref"]):
                self.assertNotIn(private,json.dumps(good))
            for changed in ({"issue":"../escape"},{"request_id":"unsafe/key"},{"version":"bad"}):
                bad=deepcopy(suite);bad["cases"][0]["input"].update(changed)
                rejected=review(spec,bad,bindings,doc,digest(doc))
                self.assertEqual(rejected["counts"]["blocked_routes"],1)
                self.assertFalse(rejected["all_cases_match_offline"])
            stale=deepcopy(bindings);stale["bindings"][0]["tool_schema_sha256"]="f"*64
            self.assertEqual(review(spec,suite,stale,doc,digest(doc))["counts"]["blocked_routes"],1)
            disabled=deepcopy(bindings);disabled["enabled"]=False
            self.assertEqual(review(spec,suite,disabled,doc,digest(doc))["reason_route_counts"],
                             {"offline_preparation_requirements":1})
            with self.assertRaises(ImplementationError):
                review(spec,suite,bindings,doc,"f"*64)
            with self.assertRaises(ImplementationError):
                review(spec,suite,{**bindings,"tool_server_url":"https://other.invalid"},doc,digest(doc))
            mixed=deepcopy(suite)
            mixed["cases"].append({"id":"case_2","category":"missing_facts","input":{},
                                   "expected":{"status":"needs_clarification","accepted_routes":[],
                                               "review_confirmed":True}})
            result=review(spec,mixed,bindings,doc,digest(doc))
            self.assertFalse(result["all_cases_match_offline"])
            self.assertEqual(result["counts"]["clarification_cases"],1)
        self.assertFalse(self.calls)
        self.assertFalse(self.resolved)

                                     "review_confirmed":True}}]}

            self.assertEqual(fields["expected_version"]["best_case_support"],1)
            self.assertEqual(fields["issue"]["full_support_paths"],["/issue","/request_id"])
            self.assertNotIn("/unsafe",fields["request_id"]["full_support_paths"])
            self.assertFalse(by_id["resolve"]["binding_authorized"])
            self.assertFalse(result["combined_request_validated"])
            self.assertFalse(result["execution_authorized"])
            self.assertEqual(result["review_sha256"],digest({k:v for k,v in result.items() if k!="review_sha256"}))
            for private in ("SYNTHETIC_PRIVATE_RECORD","synthetic-idempotency-key",case["record_ref"]):
                self.assertNotIn(private,json.dumps(result))
            with self.assertRaises(ImplementationError):
                review(spec,suite,doc,"f"*64,selection)
            with self.assertRaises(ImplementationError):
                review(spec,suite,doc,digest(doc),{"assignments":{"done":"read_case"}})
        self.assertFalse(self.calls)
        self.assertFalse(self.resolved)

    async def test_disabled_http_proposal_from_exact_case_review(self):
        from suggest_http_fields import review
        from propose_http_bindings import propose
        spec,plan,case,bindings,_,doc,ops=self.data
        suite={"format":"openworkgraph.reviewed-shadow-cases.v1",
               "process_reference":bindings["process_reference"],"spec_sha256":digest(spec),
               "reviewed_by_customer":True,"case_inputs_authorized":True,
               "cases":[{"id":"case_1","category":"normal","input":deepcopy(case),
                         "expected":{"status":"proposal","accepted_routes":[plan["activity_ids"]],
                                     "review_confirmed":True}}]}
        selection={"assignments":{"resolve":"resolve_case","readback":"read_case"}}
        choices={"argument_choices":{"resolve":{"issue":"/issue","expected_version":"/version",
                                                "request_id":"/request_id"},
                                      "readback":{"issue":"/issue"}}}
        with patch("execute_http.DirectClient",side_effect=AssertionError("Unexpected transport")):
            report=review(spec,suite,doc,digest(doc),selection)
            result=propose(spec,suite,doc,digest(doc),selection,report,report["review_sha256"],choices)
            self.assertFalse(result["enabled"])
            self.assertEqual(len(result["bindings"]),2)
            self.assertTrue(all(b["authorized"] is False and b["operation_kind"]=="unmapped" for b in result["bindings"]))
            self.assertEqual(result["proposal_metadata"]["combined_request_checks"],
                             [{"activity_id":"readback","combined_request_checks_passed":1},
                              {"activity_id":"resolve","combined_request_checks_passed":1}])
            self.assertFalse(result["proposal_metadata"]["execution_authorized"])
            self.assertNotIn(case["issue"],json.dumps(result))
            self.assertNotIn(case["request_id"],json.dumps(result))
            changed=deepcopy(report);changed["clarification_cases"]=99
            with self.assertRaises(ValueError):
                propose(spec,suite,doc,digest(doc),selection,changed,report["review_sha256"],choices)
            with self.assertRaises(ValueError):
                propose(spec,suite,doc,digest(doc),selection,report,"f"*64,choices)
            invalid=deepcopy(choices);invalid["argument_choices"]["resolve"]["issue"]="/version"
            # Integer versions do not satisfy the string path argument.
            with self.assertRaises(ValueError):
                propose(spec,suite,doc,digest(doc),selection,report,report["review_sha256"],invalid)
        self.assertFalse(self.calls)
        self.assertFalse(self.resolved)
