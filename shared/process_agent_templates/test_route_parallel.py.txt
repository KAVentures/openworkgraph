"""Token-flow route search tests for releases containing the portable model."""
import unittest
from execute import digest


class ParallelRouteSearch(unittest.TestCase):
    def test_parallel_schedules_remain_distinct(self):
        from owg_model.process_model import ProcessModel
        from route_candidates import _search
        model=ProcessModel.model_validate({'goal':'Parallel synthetic review',
            'baseline_path':['Start','A','B','Done'],'baseline_source':'user_declared',
            'baseline_coverage':'user_declared_design',
            'activities':[{'id':'start','label':'Start','kind':'start'},
                {'id':'split','label':'Fork','kind':'parallel_split'},
                {'id':'a','label':'A'},{'id':'b','label':'B'},
                {'id':'join','label':'Join','kind':'parallel_join'},
                {'id':'done','label':'Done','kind':'end'}],
            'transitions':[{'source':a,'target':b} for a,b in [('start','split'),('split','a'),('split','b'),
                ('a','join'),('b','join'),('join','done')]],
            'entry_activity_id':'start','terminal_activity_ids':['done']}).model_dump()
        spec={'revision':1,'workflow_id':'parallel','content_sha256':'a'*64,'process_model':model,
            'process_model_sha256':digest(model),'execution_authorization':False,'live_business_tools':[],
            'control_flow_contract':'openworkgraph.parallel-node-firing-plan.v1',
            'execution_schedule':'serial_token_valid_node_order'}
        result=_search(spec,[],[])
        self.assertEqual(result['status'],'competing_supported_routes')
        self.assertEqual(result['supported_schedule_count'],2)
        self.assertTrue(result['search_complete'])
        self.assertIsNone(result['unique_activity_ids'])
        from owg_model.process_parallel import replay_node_ids
        from review_facts import _scenario_coverage
        from unittest.mock import patch
        path=result['examples'][0]['activity_ids']
        replay=replay_node_ids(path,ProcessModel.model_validate(model))
        self.assertEqual(replay['transition_indices'],list(range(6)))
        self.assertEqual(replay['guaranteed_transition_indices'],list(range(6)))
        # Deliberately supply a replay projection in which one possible edge
        # is not guaranteed, to exercise the strict coverage consumer.
        case={'id':'case_1','expected':{'accepted_routes':[path]}}
        row={'case_id':'case_1','candidate_examples':[{'activity_ids':path,'status':'supported'}],
            'search_complete':True,'search_status':'unique_supported_route',
            'matches_reviewed_expectation':True,'decision_checks':[]}
        with patch('owg_model.process_parallel.replay_node_ids',return_value={
                'transition_indices':list(range(6)),'guaranteed_transition_indices':list(range(5))}):
            coverage=_scenario_coverage(spec,[case],[row])
        self.assertFalse(coverage['all_declared_items_have_reviewed_matching_support'])
        omitted=model['transitions'][5]
        item=next(e for e in coverage['transitions'] if
                  (e['source_activity_id'],e['target_activity_id'])==(omitted['source'],omitted['target']))
        self.assertEqual(item['possible_reviewed_matching_cases'],1)
        self.assertEqual(item['reviewed_matching_cases'],0)
