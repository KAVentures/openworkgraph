"""Synthetic route rehearsal fixtures with no business tools or provider calls."""
import json
import unittest
from copy import deepcopy
from datetime import datetime,timezone

from execute import ImplementationError,digest,reference
from review_bindings import rehearse
from test_execution import fixture


def inputs():
    spec,plan,case,bindings,_=fixture()
    suite={"format":"openworkgraph.reviewed-shadow-cases.v1",
        "process_reference":reference(spec),"spec_sha256":digest(spec),
        "reviewed_by_customer":True,"case_inputs_authorized":True,
        "cases":[{"id":"case_1","category":"normal","input":case,
            "expected":{"status":"proposal","accepted_routes":[plan["activity_ids"]],
                        "review_confirmed":True}}]}
    def tool(name,fields):
        return {"name":name,"description":"untrusted description",
                "input_schema_sha256":digest(fields),
                "input_fields":[{"name":n,"type":k,"required":True} for n,k in fields.items()],
                "schema_constraints_omitted":True,"annotations_are_authorization":False}
    tools=[tool("resolve_case",{"issue":"string","expected_version":"integer","request_id":"string"}),
           tool("read_case",{"issue":"string"})]
    catalog={"format":"openworkgraph.customer-tool-catalog.v1","process_reference":reference(spec),
        "spec_sha256":digest(spec),"tool_server_url":bindings["tool_server_url"],
        "inspected_at":datetime.now(timezone.utc).isoformat(),"tools":tools}
    pins={t["name"]:t["input_schema_sha256"] for t in tools}
    for row in bindings["bindings"]:
        row["tool_schema_sha256"]=pins[row["tool_name"]]
        if row.get("precondition"):
            row["precondition"]["tool_schema_sha256"]=pins[row["precondition"]["tool_name"]]
    return spec,suite,bindings,catalog


class RehearsalCases(unittest.TestCase):
    def test_reviewed_mapping_matches_metadata_without_disclosing_values(self):
        spec,suite,bindings,catalog=inputs()
        result=rehearse(spec,suite,bindings,catalog)
        self.assertEqual(result["counts"]["offline_checks_match_routes"],1)
        self.assertEqual(result["activity_ids_without_reviewed_route"],[])
        self.assertEqual(result["cases"][0]["routes"][0]["deferred_live_checks"]["terminal_source_check_count"],1)
        self.assertFalse(result["execution_authorized"])
        self.assertFalse(result["full_json_schema_validated"])
        self.assertNotIn("SYNTHETIC_PRIVATE_RECORD",json.dumps(result))
        self.assertNotIn("synthetic-idempotency-key",json.dumps(result))

    def test_missing_schema_pin_and_required_field(self):
        spec,suite,bindings,catalog=inputs()
        bindings["bindings"][0].pop("tool_schema_sha256")
        bindings["bindings"][0]["argument_map"].pop("issue")
        result=rehearse(spec,suite,bindings,catalog)
        self.assertEqual(result["counts"]["blocked_routes"],1)
        issues=result["cases"][0]["routes"][0]["requirements"]
        self.assertIn("schema_pin_missing",issues)
        self.assertIn("required_argument_missing",issues)

    def test_type_mismatch_and_missing_tool(self):
        spec,suite,bindings,catalog=inputs()
        catalog["tools"][0]["input_fields"][0]["type"]="boolean"
        catalog["tools"].pop()
        result=rehearse(spec,suite,bindings,catalog)
        issues=result["cases"][0]["routes"][0]["requirements"]
        self.assertIn("argument_type_mismatch",issues)
        self.assertIn("tool_not_in_inspected_catalog",issues)

    def test_disabled_binding_and_clarification_not_passed(self):
        spec,suite,bindings,catalog=inputs()
        bindings["enabled"]=False
        suite["cases"].append({"id":"case_2","category":"missing_facts","input":{},
            "expected":{"status":"needs_clarification","accepted_routes":[],"review_confirmed":True}})
        result=rehearse(spec,suite,bindings,catalog)
        self.assertEqual(result["counts"]["blocked_routes"],1)
        self.assertEqual(result["counts"]["clarification_only_cases"],1)
        self.assertIn("customer_bindings_disabled",result["requirement_route_counts"])

    def test_catalog_endpoint_and_staleness_refused(self):
        spec,suite,bindings,catalog=inputs()
        with self.assertRaises(ImplementationError):
            rehearse(spec,suite,bindings,{**catalog,"tool_server_url":"https://other.example/mcp"})
        with self.assertRaises(ImplementationError):
            rehearse(spec,suite,bindings,{**catalog,"inspected_at":"2000-01-01T00:00:00Z"})

    def test_preparation_error_is_minimized(self):
        spec,suite,bindings,catalog=inputs()
        del suite["cases"][0]["input"]["issue"]
        result=rehearse(spec,suite,bindings,catalog)
        self.assertEqual(result["cases"][0]["routes"][0]["requirements"],["offline_prepare_rejected"])
